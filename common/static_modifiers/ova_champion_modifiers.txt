### OVA shared champion layer - modifiers
# ova_champion_patronage is a MARKER the monthly tick reads: while present it
# adds Renown points (NOT prestige - the Royal Patronage button is a separate
# toggle). Storyline decisions apply it with a duration. Kept effect-free so
# its only in-game consequence is the scripted Renown gain.

ova_champion_patronage = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_coins_positive.dds
}

# Toggle markers for the two upkeep programs (Royal Patronage / Closed-Door
# Training). The button adds/removes the modifier - checked with has_modifier
# instead of a variable so the program shows up in the country modifier list
# and the budget. The GDP-scaled monthly cost is still billed in the tick; the
# flat country_expenses_add here is the visible budget line.
ova_champion_patronage_program = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_coins_negative.dds
	country_expenses_add = 1
}

ova_champion_training_program = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_coins_negative.dds
	country_expenses_add = 1
}

# Re-applied monthly with multiplier = champion renown (0..100)
ova_champion_renown = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_statue_positive.dds
	country_prestige_add = 1
}
