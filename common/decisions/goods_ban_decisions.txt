# Isekai TC: migrated from vanilla common/decisions/goods_ban_decisions.txt (2026-07-11).
# Generic liquor-prohibition decisions kept (hooks common/journal_entries/00_prohibition_laws.txt
# which is itself deferred, but these two decisions are self-contained and safe to enable).
# Dropped remove_opium_ban: tied to China-specific je_opium_obsession/china_welfare modifier
# with no mod analogue.

ban_liquor = {
	is_shown = {
		has_variable = prohibition_laws_var
		exists = market
		market.owner = root
		NOT = { is_banning_goods = g:liquor }
	}

	possible = {
		NOT = { has_law_or_variant = law_type:law_free_trade }
		exists = market
	}

	when_taken = {
		add_banned_goods = g:liquor
		add_modifier = {
			name = liqour_ban_authority
		}
	}

    ai_chance = {
		value = 0
		if = {
			limit = {
				OR = {
					has_strategy = ai_strategy_conservative_agenda
					has_strategy = ai_strategy_reactionary_agenda
					# removed: vanilla ref 'ai_strategy_maintain_mandate_of_heaven' (China-specific) not present in this mod
				}
			}
			add = 10
		}
    }
}

remove_liquor_ban = {
	is_shown = {
		is_banning_goods = g:liquor
		exists = market
		market.owner = root
	}

	possible = {
		always = yes
	}

	when_taken = {
		remove_banned_goods = g:liquor
		remove_modifier = liqour_ban_authority
	}

   ai_chance = {
      value = 0
		if = {
			limit = {
				has_strategy = ai_strategy_progressive_agenda
			}
			add = 10
		}
   }
}
