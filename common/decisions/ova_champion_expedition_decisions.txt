### Champions JE - AI-only expedition-into-decentralized decision
#
# The engine never evaluates diplomatic ACTIONS (like ova_champion_expedition)
# against decentralized-country targets: the AI's diplomatic-action pass simply
# skips decentralized nations. That means an AI champion nation would never
# launch an expedition into the exact targets expeditions are most flavourful
# against (unrecognized border tribes, monster marches, etc.).
#
# This decision is the workaround: it is AI-only (is_player = no), and when
# taken it hand-picks a reachable neighboring decentralized nation and forges
# the ova_champion_expedition pact against it directly with create_diplomatic_pact.
# Because create_diplomatic_pact bypasses the action's own accept_effect, this
# decision replicates that effect inline (set ova_expedition_active, switch off
# the two upkeep programs). Human players use the diplomatic action itself, which
# CAN target decentralized nations from the UI, so no player-facing decision is
# needed.

ova_champion_expedition_decentralized_decision = {
	is_shown = {
		is_player = no
		ova_champion_is_available = yes
		NOT = { ova_champion_is_adventuring = yes }
		NOT = { has_variable = ova_expedition_lockout }
		# a champion seasoned enough to send abroad, mirroring the action's own
		# will_propose level floor for non-crusade targets
		var:ova_champion_level >= 10
		# do not divert the champion while a storyline that already claims them
		# is running: the murim Returner arc, the demonic Blood Lotus sect arc,
		# or the radiant Eternal Crusade
		NOR = {
			has_journal_entry = je_ova_murim_sect_banner
			has_journal_entry = je_ova_murim_alliance_banner
			has_journal_entry = je_ova_murim_heavenly_demon
			has_journal_entry = je_ova_xianxia_blood_lotus
			has_journal_entry = je_ova_eternal_crusade
		}
	}

	possible = {
		# at least one reachable decentralized neighbour. A Champion Sanctuary
		# treaty is not pre-checked here: decentralized nations do not hold such
		# treaties, and the pact's own requirement_to_maintain sanctuary trigger
		# would auto-break the expedition on the off chance one existed.
		any_country = {
			is_country_type = decentralized
			is_adjacent_to_country = root
		}
	}

	when_taken = {
		random_country = {
			limit = {
				is_country_type = decentralized
				is_adjacent_to_country = root
			}
			save_scope_as = ova_expedition_dc_target
		}
		if = {
			limit = { exists = scope:ova_expedition_dc_target }
			# --- inline copy of ova_champion_expedition's accept_effect, because
			#     create_diplomatic_pact does not run the action's accept hook ---
			set_variable = ova_expedition_active
			if = {
				limit = { has_modifier = ova_champion_patronage_program }
				remove_modifier = ova_champion_patronage_program
			}
			if = {
				limit = { has_modifier = ova_champion_training_program }
				remove_modifier = ova_champion_training_program
			}
			create_diplomatic_pact = {
				country = scope:ova_expedition_dc_target
				type = ova_champion_expedition
			}
		}
	}

	ai_chance = {
		value = 15
		# stronger champions campaign more readily
		if = {
			limit = { var:ova_champion_level >= 30 }
			add = 15
		}
		# don't bankrupt ourselves on upkeep
		if = {
			limit = { scaled_gold_reserves < 0.1 }
			multiply = 0
		}
	}
}
