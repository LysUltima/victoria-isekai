### Champions JE - Champion Hall production methods
# The active PMs are read by ova_champion_monthly_tick: the Martial and
# Scholarly lines grant XP (tier 1/2/3/4 = +1/+2/+3/+4), the Patronage line
# grants renown points (+1/+2/+3/+4). That script-driven growth is invisible to
# the engine's auto-tooltip AND PMs have no _desc field, so each active PM
# carries a no-effect boolean display modifier (country_ova_champion_xp_N_bool /
# _renown_N_bool, defined in modifier_type_definitions/15_ova_champion_modifier_types.txt)
# as its CUSTOM TOOLTIP - the confirmed vanilla pattern (cf. state_control_strait_bool).
# Inputs are EXPONENTIAL (lead good doubles each tier: 10 -> 20 -> 40 -> 80) and
# more VARIED each tier (a new good folded in).

pm_ova_champion_hall = {
	texture = "gfx/interface/icons/production_method_icons/scholastic_education.dds"

	building_modifiers = {
		workforce_scaled = {
			goods_output_services_add = 5
		}

		level_scaled = {
			building_employment_bureaucrats_add = 250
		}
	}

    country_modifiers = {
        workforce_scaled = {
            country_weekly_innovation_add = 0.5
        }
    }
}
