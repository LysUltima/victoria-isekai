is_country_on_same_continent = {
	custom_description = {
		text = "is_country_on_same_continent"
		subject = this
		object = $TARGET$

		trigger_if = {
			limit = {
				this.capital.state_region = {
					region = sr:region_isekai
				}
			}
			$TARGET$.capital.state_region = {
				region = sr:region_isekai
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					always = no # geographic_region_americas (no mod analogue)
				}
			}
			$TARGET$.capital.state_region = {
				always = no # geographic_region_americas (no mod analogue)
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					always = no # geographic_region_africa (no mod analogue)
				}
			}
			$TARGET$.capital.state_region = {
				always = no # geographic_region_africa (no mod analogue)
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					OR = {
						region = sr:region_xianxia
						region = sr:region_xianxia_yao
						region = sr:region_xianxia_diabolic
					}
				}
			}
			$TARGET$.capital.state_region = {
				OR = {
					region = sr:region_xianxia
					region = sr:region_xianxia_yao
					region = sr:region_xianxia_diabolic
				}
			}
		}
	}
}
