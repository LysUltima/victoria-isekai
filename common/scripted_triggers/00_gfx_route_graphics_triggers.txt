						any_primary_culture = {
							OR = {
								has_discrimination_trait_group = heritage_group_desert_human
								has_discrimination_trait_group = heritage_group_xianxia_human
								has_discrimination_trait_group = heritage_group_xianxia_human
							}
						}
					}
				}
			}
		}
		power_bloc ?= {
			power_bloc_leader = {
						any_primary_culture = {
					OR = {
						has_discrimination_trait_group = heritage_group_desert_human
						has_discrimination_trait_group = heritage_group_xianxia_human
						has_discrimination_trait_group = heritage_group_xianxia_human
					}
				}
			}
		}
	}
}
