# (SWO/HEA/FLO/PUR/MYS/FEA/SPI/VER/SNO) - shared scripted triggers.

# Verified vanilla patterns used here:

#    common/journal_entries/00_german_unification.txt, common/scripted_triggers/00_clothes_triggers.txt
#  - `calc_true_if = { amount >= N ... }`: common/journal_entries/05_austria_journal_entries.txt,
#    common/scripted_triggers/ova_murim_triggers.txt (ova_murim_cult_owns_five_capitals_trigger)

#    ova_victoria_char pattern (common/journal_entries/ova_hero_journal.txt je_ova_victoria_1)
#  - `any_country = { NOT = { THIS = ROOT } ... }`: common/ai_strategies/02_diplomatic_strategies.txt

# Any of the 6 Great Feng vassal sects or 3 Great Yao vassal sects - THIS = one of the 9
# righteous tags.
ova_xianxia_is_righteous_sect_trigger = {
	OR = {
		c:SWO ?= THIS
		c:HEA ?= THIS
		c:FLO ?= THIS
		c:PUR ?= THIS
		c:MYS ?= THIS
		c:FEA ?= THIS
		c:SPI ?= THIS
		c:VER ?= THIS
		c:SNO ?= THIS
	}
}

# The 4 demonic-path sects of the Blood Lotus Covenant - THIS = BLO/GHO/SHA/NIN.
ova_xianxia_is_demonic_sect_trigger = {
	OR = {
		c:BLO ?= THIS
		c:GHO ?= THIS
		c:SHA ?= THIS
		c:NIN ?= THIS
	}
}

# Any of the 13 sects that may host a Destined One (righteous or demonic path).
ova_xianxia_is_sect_trigger = {
	OR = {
		ova_xianxia_is_righteous_sect_trigger = yes
		ova_xianxia_is_demonic_sect_trigger = yes
	}
}

# The 13 sects plus the two imperial courts FEN and YAO - all war-layer participants
# (FEN/YAO never host a Destined One, but they take part in the dao war).
ova_xianxia_is_participant_trigger = {
	OR = {
		ova_xianxia_is_sect_trigger = yes
		c:FEN ?= THIS
		c:YAO ?= THIS
	}
}

# THIS fights for the demonic path.
ova_xianxia_on_demonic_side_trigger = {
	ova_xianxia_is_demonic_sect_trigger = yes
}

# THIS fights for the righteous path - the 9 sects plus the two imperial courts.
ova_xianxia_on_righteous_side_trigger = {
	OR = {
		ova_xianxia_is_righteous_sect_trigger = yes
		c:FEN ?= THIS
		c:YAO ?= THIS
	}
}

# True while at least one demonic-path tag still stands as an independent, existing country.
ova_xianxia_demonic_side_still_stands_trigger = {
	OR = {
		AND = { exists = c:BLO c:BLO = { is_subject = no } }
		AND = { exists = c:GHO c:GHO = { is_subject = no } }
		AND = { exists = c:SHA c:SHA = { is_subject = no } }
		AND = { exists = c:NIN c:NIN = { is_subject = no } }
	}
}

# The demonic path's victory: Great Feng is gone or bowed to the Covenant, OR the demonic path
# holds 5 of the 9 righteous capitals. Scope-agnostic - safe to call from any country scope.
ova_xianxia_demonic_victory_trigger = {
	ova_xianxia_demonic_side_still_stands_trigger = yes
	OR = {
		NOT = { exists = c:FEN }
		c:FEN ?= {
			OR = {
				AND = { exists = c:BLO is_subject_of = c:BLO }
				AND = { exists = c:GHO is_subject_of = c:GHO}
				AND = { exists = c:SHA is_subject_of = c:SHA }
				AND = { exists = c:NIN is_subject_of = c:NIN }
			}
		}
		ova_xianxia_demonic_owns_five_capitals_trigger = yes
	}
}

# The righteous path's victory: the Blood Lotus Sect (the Covenant's head) is gone, or is itself
# a subject of anyone outside the demonic path.
ova_xianxia_righteous_victory_trigger = {
	OR = {
		NOT = { exists = c:BLO }
		c:BLO ?= {
			is_subject = yes
			NOT = { AND = { exists = c:GHO is_subject_of = c:GHO} }
			NOT = { AND = { exists = c:SHA is_subject_of = c:SHA } }
			NOT = { AND = { exists = c:NIN is_subject_of = c:NIN } }
		}
	}
}

# The Destined One hosted by THIS is alive - a champion character var was captured for it and
# still resolves. Set lazily by je_ova_xianxia_mc_1's on_monthly_pulse (create_character is
# deferred, so the arrival effect cannot save it directly).
ova_xianxia_mc_is_alive_trigger = {
	has_variable = ova_xianxia_mc_char
	exists = var:ova_xianxia_mc_char
}

# ROOT may send its champion on a xianxia outing right now: a living, available champion, not
# currently on the shared expedition, and not on the shared hinterland/outing cooldown.
ova_xianxia_outing_ready_trigger = {
	ova_champion_is_available = yes
	NOT = { ova_champion_is_adventuring = yes }
	NOT = { has_variable = ova_adventure_cooldown }
}

# True if ANY other sect currently hosts a living Destined One of its own - our rival pool for
# the Duel of Destinies chain.
ova_xianxia_rival_exists_trigger = {
	any_country = {
		NOT = { THIS = ROOT }
		has_variable = ova_xianxia_mc_type
		ova_xianxia_mc_is_alive_trigger = yes
		ova_has_living_champion = yes
	}
}
