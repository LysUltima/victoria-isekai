### Isekai storyline - "War of the Dawn" diplomatic play

# annex_as_civil_war = yes): the old Lysnan crown against Victoria's Concord

# hold ova_restoration against each other. Structure follows vanilla
# dp_revolution, but unlike a vanilla civil war this one ALLOWS a negotiated
# peace: since the only war goal on the table is the full annexation, the sole
# soft option the negotiation offers is a white peace (status quo), letting the
# two crowns lay down arms and live sundered rather than fight to the death.

	enable_switch_sides = yes

	mirror_war_goal = yes
	initiator_can_add_war_goals = no
