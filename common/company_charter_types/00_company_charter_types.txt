investment_charter = {
	type = investment

	icon = "gfx/interface/icons/company_charter_icons/investment_charter.dds"

	cooldown = { years = 5 }

	ai_possible = {
		company_prosperity >= 50
		owner = {
			NOT = {
				has_law_or_variant = law_type:law_isolationism
			}

			OR = {
				any_scope_treaty = {
					any_scope_article = {
						has_type = foreign_investment_rights

					}
				}
				num_subjects >= 1
			}
		}
	}

	ai_weight = {
        value = 10

		if = {
			limit = {

		}

					always = no # was company_type:company_yasuda (Japan-flavor company not present in this mod)
				}
			}
			add = -20
		}

colonization_charter = {
	type = colonization

	icon = "gfx/interface/icons/company_charter_icons/colonization_charter.dds"

	additional_input = yes

	possible = {
		owner ?= {
			NOT = { is_country_type = unrecognized }
		}
	}

	ai_possible = {
		company_prosperity >= 50
		OR = {
			always = no
			always = no
			always = no
			always = no
			always = no
			always = no
			always = no
			always = no
			always = no
			always = no
			always = no
			always = no
			always = no
		}
		scope:target = {
			NOT = {
				any_scope_state = {
					OR = {
						is_homeland_of_country_cultures = root.owner
						any_neighbouring_state = {
							OR = {
								is_homeland_of_country_cultures = root.owner
								AND = {
									owner = root.owner
									is_incorporated = yes
								}
							}
						}
					}
				}
			}
		}
	}

	ai_weight = {
        value = 0

        if = {
			limit = {
				owner = {
					OR = {
						has_strategy = ai_strategy_colonial_expansion
						has_strategy = ai_strategy_plantation_economy
					}
				}
			}
			add = 10
		}

    }
}
