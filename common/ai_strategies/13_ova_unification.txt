ai_strategy_ova_cultural_unification = {
	icon = "gfx/interface/icons/ai_strategy_icons/unification.dds"
	type = diplomatic

	undesirable_infamy_level = { value = 50 }
	unacceptable_infamy_level = { value = 100 }
	ideological_opinion_effect_mult = 0.5
	wargoal_maneuvers_fraction = { value = 0.2 }
	diplomatic_play_neutrality = { value = 0 }
	diplomatic_play_boldness = { value = 50 }
	recklessness = { value = 0.5 }

	aggression = {
		value = 0.25
		if = {
			limit = { ova_is_unification_target = yes }
			add = 5
		}
	}

	building_group_weights = { bg_army = 1.25 }

	state_value = {
		if = {
			limit = {
				ova_can_pursue_cultural_unification = yes
				scope:target_state = {
					state_region = { ova_is_unification_homeland = { COUNTRY = ROOT } }
				}
				can_reach_target_state_to_conquer = { TARGET_STATE = scope:target_state }
			}
			add = { value = 100 desc = "OVA_STATE_VALUE_CULTURAL_UNIFICATION" }
		}
	}

	wargoal_scores = {
		annex_country = {
			if = {
				limit = {
					ova_is_unification_target = yes
					# Prefer whole-country annexation only when all its states count toward formation.
					scope:target_country = {
						NOT = {
							any_scope_state = {
								state_region = { NOT = { ova_is_unification_homeland = { COUNTRY = ROOT } } }
							}
						}
					}
					can_reach_target_country_to_conquer = { TARGET_COUNTRY = scope:target_country }
				}
				add = 300
			}
		}
	}

	secret_goal_scores = {
		conquer = {
			trigger = {
				ova_is_unification_target = yes
				can_reach_target_country_to_conquer = { TARGET_COUNTRY = scope:target_country }
			}
			score = { value = 500 }
		}
	}

	possible = { ova_can_pursue_cultural_unification = yes }
	weight = { value = 500 }
}
