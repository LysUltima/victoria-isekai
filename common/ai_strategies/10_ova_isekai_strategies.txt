### Isekai hero storyline - AI strategies
# These strategies adapt the vanilla territorial expansion strategy to the
# crusade and the war over the demon homeland.

ai_strategy_ova_radiant_crusader = {
	icon = "gfx/interface/icons/ai_strategy_icons/territorial_expansion.dds"

	type = diplomatic

	will_form_power_bloc = {
		always = no
	}

	undesirable_infamy_level = {
		value = 50

		if = {
			limit = {
				has_technology_researched = pan-nationalism
			}
			multiply = 1.5
		}
		else_if = {
			limit = {
				has_technology_researched = nationalism
			}
			multiply = 1.25
		}
	}

	unacceptable_infamy_level = {
		value = 100
	}

	ideological_opinion_effect_mult = {
		value = 0.75
	}

	wargoal_maneuvers_fraction = {
		value = 0.15
	}

	diplomatic_play_neutrality = {
		value = 25
	}

	diplomatic_play_boldness = {
		value = 25
	}

	recklessness = {
		value = 0.25
	}

	aggression = {
		value = 1

		if = {
			limit = {
				NOT = { is_country_type = unrecognized }
				scope:target_country = { is_country_type = unrecognized }
			}
			add = 1
		}

		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.25
		}

		# More aggressive against ideological enemies, a bit less aggressive against ideological allies
		multiply = {
			value = "ai_ideological_opinion(scope:target_country)"
			multiply = -0.1
			add = 1
			max = 2.0
			min = 0.5
		}
	}

	building_group_weights = {
		bg_army = 1.5
	}

	secret_goal_weights = {
		protect = 0.5
		conquer = 1.5
	}

	wargoal_weights = {
		annex_country = 1.5
		conquer_state = 2.0
		return_state = 1.5
		force_nationalization = 0.25
	}

	secret_goal_scores = {
		conquer = {
			trigger = {
				NOT = { scope:target_country = { is_subject_of = root } }
				NOT = { is_subject_of = scope:target_country }
				can_reach_target_country = yes
				OR = {
					c:DEM ?= scope:target_country
					AND = {
						has_strategic_adjacency = scope:target_country
						scope:target_country.country_rank <= root.country_rank
					}
				}
			}
			score = {
				if = {
					limit = {
						NOT = { is_country_type = unrecognized }
						scope:target_country = { NOT = { is_country_type = unrecognized } }
					}
					add = 25
				}
				else_if = {
					limit = {
						scope:target_country = { is_country_type = unrecognized }
					}
					add = 50
				}
			}
		}

		antagonize = {
			trigger = {
				c:DEM ?= scope:target_country
			}
			score = {
				add = 500
			}
		}

        befriend = {
            trigger = {
				c:DEM ?= scope:target_country
			}
			score = {
				subtract = 100000
			}
        }
	}

	possible = {
		ova_hero_is_radiant_crusader = yes
		exists = c:DEM
	}

	weight = {
		value = 500
	}
}

ai_strategy_ova_demonic_reclamation = {
	icon = "gfx/interface/icons/ai_strategy_icons/territorial_expansion.dds"

	type = diplomatic

	will_form_power_bloc = {
		OR = {
			global_country_ranking <= 3
			AND = {
				country_rank >= rank_value:great_power
				has_technology_researched = nationalism
			}
		}
	}

	undesirable_infamy_level = {
		value = 50

		if = {
			limit = {
				has_technology_researched = pan-nationalism
			}
			multiply = 1.5
		}
		else_if = {
			limit = {
				has_technology_researched = nationalism
			}
			multiply = 1.25
		}
	}

	unacceptable_infamy_level = {
		value = 100
	}

	ideological_opinion_effect_mult = {
		value = 0.75
	}

	wargoal_maneuvers_fraction = {
		value = 0.15
	}

	diplomatic_play_neutrality = {
		value = 25
	}

	diplomatic_play_boldness = {
		value = 25
	}

	recklessness = {
		value = 0.25
	}

	aggression = {
		value = 1

		if = {
			limit = {
				NOT = { is_country_type = unrecognized }
				scope:target_country = { is_country_type = unrecognized }
			}
			add = 1
		}

		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.25
		}

		# More aggressive against ideological enemies, a bit less aggressive against ideological allies
		multiply = {
			value = "ai_ideological_opinion(scope:target_country)"
			multiply = -0.1
			add = 1
			max = 2.0
			min = 0.5
		}

		if = {
			limit = { ova_hero_target_is_reclamation_enemy = yes }
			add = 0.5
		}

		# never against the Concord Aurorae - the dawn is a sworn friend
		# (kept last so it zeroes the reclamation bonus too: AUR holds demon
		# marches, but she holds them as a friend)
		if = {
			limit = {
				scope:target_country = { has_variable = ova_new_dawn }
			}
			multiply = 0
		}
	}

	building_group_weights = {
		bg_army = 0.75 # Otherwise the Isekai countries get murdered
	}

	secret_goal_weights = {
		protect = 0.5
		conquer = 1.5
	}

	wargoal_weights = {
		annex_country = 1.5
		conquer_state = 2.0
		# the empire holds claims on its stolen homeland states
		return_state = 2.0
		force_nationalization = 0.25
	}

	secret_goal_scores = {
        befriend = {
            trigger = {
				scope:target_country = { ova_hero_is_radiant_crusader = yes }
			}
			score = {
				subtract = 100000
			}
        }
		conquer = {
			trigger = {
				NOR = {
					is_subject_of = scope:target_country
					scope:target_country = { is_subject_of = ROOT }
					scope:target_country = { has_variable = ova_new_dawn }
				}
				can_reach_target_country = yes
				# the usual gates (adjacency, rank) do not apply to the
				# reclamation: a crown squatting on the homeland is a target
				# no matter how large it is
				OR = {
					AND = {
						has_strategic_adjacency = scope:target_country
						scope:target_country.country_rank <= root.country_rank
					}
					ova_hero_target_is_reclamation_enemy = yes
				}
			}
			score = {
				if = {
					limit = {
						NOT = { is_country_type = unrecognized }
						scope:target_country = { NOT = { is_country_type = unrecognized } }
					}
					add = 25
				}
				else_if = {
					limit = {
						scope:target_country = { is_country_type = unrecognized }
					}
					add = 50
				}
				# the reclamation's whole purpose: every state of
				# sr:region_demonland back under the Ash Throne
				if = {
					limit = { ova_hero_target_is_reclamation_enemy = yes }
					add = 100
					if = {
						limit = { scope:target_country = { ova_hero_is_radiant_crusader = yes } }
						add = 400
					}
				}
			}
		}
		# Hold a hostile attitude toward the crusaders and occupiers out of reach.
		antagonize = {
			trigger = {
				ova_hero_target_is_reclamation_enemy = yes
			}
			score = {
				value = 150
				if = {
					limit = { scope:target_country = { ova_hero_is_radiant_crusader = yes } }
					add = 150
				}
			}
		}
		# the dawn kept faith with the tribes - keep faith with the dawn
		befriend = {
			trigger = {
				scope:target_country = { has_variable = ova_new_dawn }
			}
			score = {
				value = 500
			}
		}
	}

	possible = {
		c:DEM ?= THIS
	}

	weight = {
		value = 500
	}
}
