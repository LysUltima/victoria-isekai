# Martial Sect - the state is a cultivation sect: sect elders govern, the patriarch reigns.
# Governance-principle peer of Monarchy/Theocracy for the xianxia world.

law_martial_sect = {
	group = lawgroup_governance_principles

	icon = "gfx/interface/icons/law_icons/bakufu.dds"

	progressiveness = -25

	disallowing_laws = {
		law_anarchy
	}

	can_enact = {
		trigger_if = {
			limit = {
				is_subject = no
			}
			NOT = { has_government_type = gov_chartered_company }
		}
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_governance_principles_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	on_activate = {
		transition_off_of_monarchy_law_effects = yes
	}

	modifier = {
		country_legitimacy_headofstate_add = 20
		country_legitimacy_ideological_incoherence_mult = 0.1
		interest_group_ig_armed_forces_pol_str_mult = 0.25
		country_authority_add = 200
	}

	ai_will_do = {
		exists = ruler
		ig:ig_armed_forces = {
			is_in_government = yes
			is_powerful = yes
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
