# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law

law_no_womens_rights = {
	group = lawgroup_rights_of_women
	icon = "gfx/interface/icons/law_icons/no_womens_rights.dds"

	progressiveness = -100

	on_enact = {
		recalculate_pop_ig_support = yes
	}

	is_visible = {
		NOT = {
			has_law = law_type:law_women_in_the_fields
		}
		NOT = {
			ova_country_is_matriarchal_society = yes
		}
	}

	modifier = {
		state_birth_rate_mult = 0.05
	}

	ai_will_do = {
		literacy_rate < 0.4
		exists = ruler
		ruler = {
			has_ideology = ideology:ideology_traditionalist
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

# Matriarchal branch (dark elves, beastfolk)
# Mirrors the regular line: only women rule, own, and command.
# Locked per country - a matriarchal society never sees the
# patriarchal laws and vice versa (is_visible on both branches).
############################################################

law_matriarchy = {
	group = lawgroup_rights_of_women
	parent = law_no_womens_rights # Inherits ideological stances from parent
	icon = "gfx/interface/icons/law_icons/womens_suffrage.dds"

	progressiveness = -100

	on_enact = {
		recalculate_pop_ig_support = yes
	}

	is_visible = {
		OR = {
			ova_country_is_matriarchal_society = yes
			any_primary_culture = {
				ova_culture_is_matriarchal = yes
			}
		}
	}

	modifier = {
		state_birth_rate_mult = 0.05
	}

	ai_will_do = {
		OR = {
			# A matriarchal people restoring the mother's line
			AND = {
				NOT = { ova_country_is_matriarchal_society = yes }
				any_primary_culture = {
					ova_culture_is_matriarchal = yes
				}
			}
			# Mirror of law_no_womens_rights: reactionary rollback
			AND = {
				literacy_rate < 0.4
				exists = ruler
				ruler = {
					has_ideology = ideology:ideology_traditionalist
				}
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
