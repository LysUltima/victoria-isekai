# The Academy Charter - the ACA-only variant law behind Saint Christine Academy.
#
# Saint Christine Academy is a SECULAR school with no tie to the Radiant Church: the masters
# govern, the Dean presides, and the faith holds no chair. Mechanically it is a variant law
# (parent = law_presidential_republic), which is the closest secular principle in the group: a
# head of state who is neither crowned nor ordained nor born to it, raised up by an institution
# and answerable to it. The parent only supplies the lawgroup slot and the ideologies' stances;
# the Academy is not an elected republic, which is why every government type in
# common/government_types/00_ova_academy.txt is transfer_of_power = dictatorial.
#
# It is hidden from every country except ACA, cannot be enacted, imposed or petitioned for
# anywhere else, and ACA starts with it from effect_starting_politics_academy.

law_academy = {
	group = lawgroup_governance_principles
	parent = law_presidential_republic # Inherits ideological stances from parent

	icon = "gfx/interface/icons/law_icons/technocracy.dds"

	progressiveness = 25 # Secular and open to talent, but the faculty are nobody's electorate

	# The charter is Saint Christine's alone: hidden, unenactable and unimposable anywhere else.
	is_visible = {
		c:ACA ?= ROOT
	}

	can_enact = {
		c:ACA ?= ROOT
		trigger_if = {
			limit = {
				is_subject = no
			}
			NOT = { has_government_type = gov_chartered_company }
		}
	}

	can_impose = {
		always = no
	}

	# Reduced transition (clears Bakufu only). The Dean is not an elected president, so the full
	# transition_off_of_monarchy_effects used by the parent law is deliberately not used here.
	on_activate = {
		transition_off_of_monarchy_law_effects = yes
	}

	modifier = {
		country_legitimacy_headofstate_add = 20
		country_legitimacy_ideological_incoherence_mult = 0.1
		interest_group_ig_intelligentsia_pol_str_mult = 0.25
		country_academics_pol_str_mult = 0.25
		country_authority_add = 100
		country_production_tech_spread_mult = 0.1
		country_military_tech_spread_mult = 0.1
		country_society_tech_spread_mult = 0.1
	}

	ai_will_do = {
		c:ACA ?= ROOT
		exists = ruler
	}

	ai_enact_weight_modifier = {
		value = 0
	}

	ai_impose_chance = {
		value = 0
	}
}
