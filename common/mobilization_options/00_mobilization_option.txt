mobilization_option_extra_supplies = {
	texture = "gfx/interface/icons/mobilization_options/extra_supplies.dds"

	possible = {
		market ?= {
			mg:groceries ?= {
				has_potential_supply = yes
			}
		}
	}

	unlocking_technologies = {
		army_reserves
	}

	on_deactivate_while_mobilized = {
		custom_tooltip = {
			text = mobilization_option_it_hurts_morale_when_you_remove_supplies_while_in_combat_tt
			every_combat_unit = {
				add_morale = {
					value = morale
					multiply = -0.5
				}
			}
		}
	}

	upkeep_modifier = {
		goods_input_groceries_add = 1.5
	}
	upkeep_modifier_unscaled = {
        goods_input_small_arms_mult = 0.25
        goods_input_artillery_mult = 0.25
		goods_input_ammunition_mult = 0.25
        goods_input_coal_mult = 0.25
        goods_input_telephones_mult = 0.25
        goods_input_engines_mult = 0.25
	}

	unit_modifier = {
		unit_morale_loss_mult = -0.05
		unit_offense_mult = 0.1
		unit_defense_mult = 0.1
	}
	ai_weight = {
		value = 1

		if = {
			limit = {
				always = no
				scope:military_formation ?= {
					any_combat_unit = {
						building = {
							state = {
								state_region = {
									OR = {
always = no
always = no
									}
								}
							}
						}
						percent >= 0.2
					}
				}
			}
			add = -1000000
		}
	}

	group = supplies
}
