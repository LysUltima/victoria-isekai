country_legislature_name = {
	type = country
	text = {
		trigger = {
			has_law_or_variant = law_type:law_technocracy
		}
		localization_key = legislature_board_of_directors
	}
	text = {
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_corporate_state
				has_law_or_variant = law_type:law_social_monarchy
			}
		}
		localization_key = legislature_chamber_of_corporations
	}
	text = {
		trigger = {
			country_has_voting_franchise = yes
		}
		localization_key = legislature_parliament
	}
	text = { # Fallback
		trigger = {
			country_has_voting_franchise = no
		}
		localization_key = legislature_court
	}
}
