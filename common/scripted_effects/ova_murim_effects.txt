# Heavenly Demon Sect (c:CHE) vs Orthodox Martial Alliance storyline - scripted effects.
# ALL character bookkeeping (creation, levels, popularity, injury, seclusion, death, traits,
# army buffs) is delegated to the SHARED champion layer in
# common/scripted_effects/ova_champion_effects.txt - this file only implements the Returner
# (Kang Jin-woo) dispatch/arrival/refusal wrappers, the Eight Banners courtship machinery,
# the succession endgame, and the shattering of the Cult. All effects run in COUNTRY scope
# unless noted.
#
# 2026-07-18 second rework (the hero's road): the succession chapter is GONE - the
# Returner never takes the seat until the very end of the tale. After chapter 1 the
# arc courts every banner including the host's own (je_ova_murim_ally_home comes
# first), each courtship a 4-beat chain (intro creates the companion, two TRIAL beats
# with a selfless and a selfish option each, then the pledge) under a 5-year journal
# timeout. Selfless choices cost the Returner (wounds, forfeits, no growth) and lift
# the banner instead; selfish choices make him monstrously strong and alienate that
# banner forever (its JE fails). When every banner has been courted or lost, the
# demonic cult's assassins kill the old master (.150) and the chain forks on
# ova_murim_good_path_trigger: the wedding/acclamation ending (marches, then the
# final battle by champion expedition) or the seat taken by force (war conference,
# then the scripted Reckoning war). Both endings break CHE via
# effect_ova_murim_shatter_cult, restoring the five great orthodox factions the Cult
# destroyed generations ago (Amita/Gonryun/Jongnam Sects, Moyong Clan, Gaebang).
#
# Country variables used here:
#   ova_murim_returner_host    - marks the host country (the sect he joined)
#   ova_murim_returner_char    - character var for Kang Jin-woo, set LAZILY by
#                                je_ova_murim_returner_1's on_monthly_pulse (create_character is
#                                deferred, so the arrival effect below cannot save him directly)
#   ova_murim_good_score       - +1 per selfless choice, -3 per selfish one; >= 0 at the
#                                succession means at least 75% of choices were selfless
#   ova_murim_offer_pending    - timed 90-day guard against a double offer while one is already out
#   ova_murim_declined         - permanent per-country opt-out (that country turned him away once)
#   ova_murim_fallback_months  - HWA's counter toward the "no player claimed him" fallback (.2)
#   ova_murim_seen_NN          - per-beat seen flags driving the chapter tick ladders below
#   ova_murim_ally_tried_X     - the arc already courted banner X (X = home/sor/pae/tan/
#                                nam/jeg/mud/sam/hwa; set when its JE is added)
#   ova_murim_ally_char_X      - character var for X's scripted companion (set inside the INTRO
#                                beat's create_character on_created block, with plot armor)
#   ova_murim_ally_sworn_X     - the pledge was accepted; the courtship's personal half is done
#   ova_murim_ally_failed_X    - a selfish trial choice alienated banner X (fails its JE)
#   ova_murim_ally_passed_X    - the host declined X's pledge; that banner is not courted again
#   ova_murim_reckoning_started- the succession endgame has begun (guards a double .150)
#   ova_murim_good_end         - the good ending was taken (.151/.152); shatter grants marches
#   ova_murim_restored_X       - shatter resurrected dead banner X (marks it for a march)
#   ova_murim_duel_done        - the final battle has been triggered (single shot)
#   ova_murim_duel_won         - the Heavenly Demon fell to Kang Jin-woo's blade (.82)
# Global variables: ova_murim_returner_fate - unset = pending, 1 = placed with a host, 2 = gone
#   forever (every player sect declined, including HWA itself);
#   ova_murim_reckoning_called - gates dp_ova_murim_reckoning's possible block.

# Monthly dispatcher, called from all three war JEs' on_monthly_pulse. Offers the Returner to a
# player-run Alliance sect first; if none claims him within 3 months, he defaults to Hwasan's gate.
effect_ova_murim_returner_dispatch = {
	# a player-run Orthodox sect gets the Returner offer personally
	if = {
		limit = {
			NOT = { has_global_variable = ova_murim_returner_fate }
			exists = c:CHE
			is_player = yes
			ova_murim_is_alliance_sect_trigger = yes
			NOT = { has_variable = ova_champion_char }
			NOT = { has_variable = ova_murim_offer_pending }
			NOT = { has_variable = ova_murim_declined }
		}
		set_variable = { name = ova_murim_offer_pending days = 90 }
		trigger_event = { id = ova_murim.1 days = 5 }
	}
	# no player sect claims him: he defaults to the gates of Hwasan
	if = {
		limit = {
            exists = c:HWA
			c:HWA ?= THIS
			NOT = { has_global_variable = ova_murim_returner_fate }
			exists = c:CHE
		}
        if = {
            limit = {
                has_variable = ova_murim_fallback_months
            }
            change_variable = { name = ova_murim_fallback_months add = 1 }
        }
        else = {
            set_variable = { name = ova_murim_fallback_months value = 1 }
        }
		if = {
			limit = {
				var:ova_murim_fallback_months >= 3
				NOT = { has_variable = ova_champion_char }
				NOT = { has_variable = ova_murim_offer_pending }
				NOT = { has_variable = ova_murim_declined }
			}
			set_variable = { name = ova_murim_offer_pending days = 90 }
			trigger_event = { id = ova_murim.2 days = 5 }
		}
	}
}

# The accepting host takes him in. Guarded so a multiplayer double-accept race is a silent no-op
# for whichever country's option resolves second.
effect_ova_murim_returner_arrives = {
	if = {
		limit = { NOT = { has_global_variable = ova_murim_returner_fate } }
		set_global_variable = { name = ova_murim_returner_fate value = 1 }
		remove_variable = ova_murim_offer_pending
		set_variable = ova_murim_returner_host
        set_variable = { name = ova_murim_good_score value = 0 }
		ova_champion_create_from_template = { TEMPLATE = template_kang_jinwoo LEVEL = 12 }
		add_journal_entry = { type = je_ova_murim_returner_1 }
		trigger_event = { id = ova_murim.3 days = 7 }
	}
}

# HWA itself declines the boy at its own gate - there is no further fallback, so his tale ends here.
effect_ova_murim_returner_refused_forever = {
	set_global_variable = { name = ova_murim_returner_fate value = 2 }
	remove_variable = ova_murim_offer_pending
}

# Victoria chain's effect_ova_victoria_chN_tick ladders.

# Routes a failed chapter to the right ending event: the Returner has died
# (.72, the death of Kang Jin-woo) or the Cult has mastered the murim
# around him (.73, the calamity arrives). The Victoria chain's
# effect_ova_victoria_fail_events pattern.
effect_ova_murim_fail_events = {
	if = {
		limit = {
			has_variable = ova_murim_returner_char
			NOT = { exists = var:ova_murim_returner_char }
		}
		trigger_event = { id = ova_murim.72 days = 3 }
	}
	else = {
		trigger_event = { id = ova_murim.73 days = 3 }
	}
}
