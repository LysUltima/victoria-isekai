	effect_starting_rights_of_women = yes

		activate_law = law_type:law_freedom_of_conscience

# Rights of Women: matriarchal peoples (dark elves and all beastfolk) start on the
# matriarchal branch of the law group; everyone else starts on the regular line.
effect_starting_rights_of_women = {
	if = {
		limit = {
			any_primary_culture = {
				ova_culture_is_matriarchal = yes
			}
		}
		activate_law = law_type:law_matriarchy
	}
	else = {
		activate_law = law_type:law_no_womens_rights
	}
}
