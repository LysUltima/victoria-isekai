	transfer_of_power = hereditary

	male_ruler = "RULER_TITLE_PHILOSOPHER_KING"

	female_heir = "RULER_TITLE_HEIR_APPARENT"

		has_law = law_type:law_social_monarchy

		coa_absolute_monarchy_trigger = no
		has_law_or_variant = law_type:law_technocracy

		country_is_not_united_kingdom_subject = yes
		NOT = { is_subject_type = subject_type_crown_land }
		has_gov_regency = no

	on_government_type_change = {
		change_to_hereditary = yes
	}
	on_post_government_type_change = {
		post_change_to_hereditary = yes
	}
}
