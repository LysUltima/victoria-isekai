building_government_administration = {
	building_group = bg_bureaucracy
	city_type = city

	levels_per_mesh = 50

	unlocking_technologies = {
		tech_bureaucracy
	}

	should_auto_expand = {
		occupancy >= 0.8
		OR = {
			state.tax_capacity < state.tax_capacity_usage
			owner = {
				bureaucracy < -10
			}
		}
	}

	production_method_groups = {
		pmg_base_building_government_administration
		pmg_government_administration_bureaucrat_professionalism
		pmg_government_administration_religious_administration
	}

	icon = "gfx/interface/icons/building_icons/building_government_administration.dds"

	required_construction = construction_cost_very_low

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"
}

building_homunculus_factory = {
	building_group = bg_technology
	city_type = city

	levels_per_mesh = 25

	# The AI's government-building management prunes gov-funded buildings it
	# has no hardcoded want for (this produces no bureaucracy/innovation/units),
	# regardless of ai_value - same reason all vanilla monuments carry this.
	downsizeable = no

	unlocking_technologies = {
		homunculus_creation
	}

	production_method_groups = {
		pmg_base_building_homunculus_factory
	}

	icon = "gfx/interface/icons/building_icons/soulwrought_assembly_plant.dds"

	required_construction = construction_cost_medium

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"

	ai_value = {
		value = 2500

		if = {
			limit = { state_unemployment_rate < 0.05 }
			add = 1000000
		}
		if = {
			limit = { state_unemployment_rate < 0.02 }
			add = 4000000
		}

        if = {
            limit = { 
                owner = {
                    any_primary_culture = {
                        has_discrimination_trait_group = heritage_group_desert_human
                    }
                }
            }
            add = 5000
        }
	}
}
