ig_industrialists = {
	color = hsv{ 0.06 0.60 0.89 }
	texture = "gfx/interface/icons/ig_icons/industrialists.dds"
	layer = "revolution_dynamic_industrialists"
	index = 2

	ideologies = {
		ideology_plutocratic
		ideology_laissez_faire
		ideology_individualist
	}

	character_ideologies = {
		ideology_moderate
		ideology_market_liberal
		ideology_protectionist
	}

	enable = {
		always = yes
	}

	# Deprecated; use on_enable effect to assign traits instead
	traits = {
		ig_trait_job_creators
		ig_trait_engines_of_progress
		ig_trait_tax_avoidance
	}

	on_enable = {
		ig:ig_industrialists ?= {
			if = {
				limit = {
					owner = { always = no} # vanilla ref 'c:JAP' not present in this mod
				}
				set_interest_group_name = ig_gosho
				remove_ideology = ideology_laissez_faire
				add_ideology = ideology_interventionist
				set_ig_trait = ig_trait:ig_trait_zaibatsu_withdrawal
				set_ig_trait = ig_trait:ig_trait_engines_of_progress
				set_ig_trait = ig_trait:ig_trait_zaibatsu_cooperation
			}
			else_if = {
				limit = {
					owner = { always = no} # vanilla ref 'c:SWE' not present in this mod
				}
				set_ig_trait = ig_trait:ig_trait_tax_avoidance
				set_ig_trait = ig_trait:ig_trait_engines_of_progress
				set_ig_trait = ig_trait:ig_trait_kommerskollegium
			}
			else_if = {
				limit = {
					has_dlc_feature = ip3_content
					OR = {
						owner = { always = no} # vanilla ref 'c:NOR' not present in this mod
						owner = { always = no} # vanilla ref 'c:GRE' not present in this mod
					}
				}
				set_ig_trait = ig_trait:ig_trait_tax_avoidance
				set_ig_trait = ig_trait:ig_shipping_magnates
				set_ig_trait = ig_trait:ig_trait_job_creators
			}
			else_if = {
				limit = {
					has_dlc_feature = voice_of_the_people_content
					owner = { always = no} # vanilla ref 'c:FRA' not present in this mod
				}
				set_ig_trait = ig_trait:ig_trait_tax_avoidance
				set_ig_trait = ig_trait:ig_trait_engines_of_progress
				set_ig_trait = ig_trait:ig_trait_ventilate_unify_beautify
			}
			else_if = {
				limit = {
					has_dlc_feature = rp1_content
					owner = { always = no} # vanilla ref 'c:BRZ' not present in this mod
				}
				set_ig_trait = ig_trait:ig_trait_tax_avoidance
				set_ig_trait = ig_trait:ig_trait_the_goods_must_flow
				set_ig_trait = ig_trait:ig_trait_job_creators
			}
			# India
			else_if = {
				limit = {
					owner = {
						always = no # vanilla ref 'c:BIC' not present in this mod
					}
				}
				set_interest_group_name = ig_east_india_company
				remove_ideology = ideology_laissez_faire
				add_ideology = ideology_colonialist
				remove_ideology = ideology_plutocratic
				add_ideology = ideology_enclavist
				remove_ideology = ideology_individualist
				add_ideology = ideology_malthusian

				set_ig_trait = ig_trait:ig_trait_wounded_honor
				set_ig_trait = ig_trait:ig_trait_stamped_and_signed
				set_ig_trait = ig_trait:ig_trait_responsible_government
			}
			else_if = {
				limit = {
					owner = {
						OR = {
							always = no # vanilla ref 'c:DEI' not present in this mod
							always = no # vanilla ref 'c:ALK' not present in this mod
							always = no # vanilla ref 'c:HBC' not present in this mod
							is_country_type = company
						}
					}
				}
				remove_ideology = ideology_laissez_faire
				add_ideology = ideology_colonialist

				set_ig_trait = ig_trait:ig_trait_tax_avoidance
				set_ig_trait = ig_trait:ig_trait_engines_of_progress
				set_ig_trait = ig_trait:ig_trait_job_creators
			}
			else = {
				set_ig_trait = ig_trait:ig_trait_tax_avoidance
				set_ig_trait = ig_trait:ig_trait_engines_of_progress
				set_ig_trait = ig_trait:ig_trait_job_creators
			}
		}
	}

	on_disable = {}

	priority_cultures = { #Country scope
		rule = {
			trigger = {
				always = no # vanilla ref 'c:BIC' not present in this mod
				any_primary_culture = {
					cu:aldrican = this # was cu:british
				}
			}
			cultures = {}
		}
		rule = {
			trigger = {
				always = no # vanilla ref 'c:ZAN' not present in this mod
				any_primary_culture = {
					cu:zaharin = this # was cu:bedouin
				}
			}
			cultures = {}
		}
		rule = {
			trigger = {
				always = no # vanilla ref 'c:HAW' not present in this mod
				always = no
			}
			cultures = {}
		}
		rule = {
			trigger = {
				always = no # vanilla ref 'c:PHI' not present in this mod
				NOT = { always = no} # vanilla ref 'je_philippines_main_reliance_happened' not present in this mod
				NOT = { any_primary_culture = { cu:melati = this } }
				NOT = { any_primary_culture = { cu:melati = this } }
				any_primary_culture = { cu:alena = this} # was cu:spanish
			}
			cultures = {}
		}
		rule = {
			trigger = {
				always = no # vanilla ref 'c:PHI' not present in this mod
				any_primary_culture = {
					AND = {
						cu:alena = this # was cu:spanish
						cu:melati = this # was cu:filipino_hispanophone
					}
				}
			}
			cultures = {}
		}
	}

	on_character_ig_membership = {
		if = {
			limit = {
				OR = {
					culture = cu:aldrican # was cu:british
					culture = cu:vardish # was cu:north_german
				}
				owner = {
					always = no # vanilla ref 'c:ZAN' not present in this mod
					any_primary_culture = {
						cu:zaharin = this # was cu:bedouin
					}
				}
			}
			change_character_religion = rel:light_reformed # was rel:protestant
		}
		if = {
			limit = {
				always = no # was culture = cu:gujarati (no mod analogue)
				owner = {
					always = no # vanilla ref 'c:ZAN' not present in this mod
					any_primary_culture = {
						cu:zaharin = this # was cu:bedouin
					}
				}
			}
			change_character_religion = rel:farvahar # was rel:shiite
		}
	}

	pop_potential = {
		OR = {
			is_pop_type = capitalists
			is_pop_type = engineers
			is_pop_type = shopkeepers
			AND = {
				always = no # vanilla ref 'c:BIC' not present in this mod
				always = no # vanilla ref 'c:BIC' not present in this mod
				OR = {
					culture ?= {
						has_discrimination_trait_group = heritage_group_european
					}
					is_pop_type = bureaucrats
					is_pop_type = clerks
				}
			}
		}
	}

	pop_weight = {
		value = 0

		add = {
			desc = "POP_CAPITALISTS"
			if = {
				limit = {
					is_pop_type = capitalists
				}
				value = 250
				add = this.standard_of_living
			}
		}

		add = {
			desc = "POP_ENGINEERS"
			if = {
				limit = {
					is_pop_type = engineers
				}
				value = 50
				add = {
					value = this.standard_of_living
					multiply = 2
				}
			}
		}

		add = {
			desc = "POP_SHOPKEEPERS"
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				value = 25
				add = {
					value = this.standard_of_living
					multiply = 2.5
				}
			}
		}
		if = {
			limit = {
				owner = {
					always = no # vanilla ref 'c:BIC' not present in this mod
				}
				culture = {
					has_discrimination_trait_group = heritage_group_european
				}
				OR = {
					strata = upper
					strata = middle
				}
				NOT = {
					is_pop_type = officers
				}
			}
			add = {
				desc = "POP_INDIA_EUROPEAN_ADMINISTRATORS"
				value = 250
			}
		}

		if = {
			limit = {
				owner = {
					always = no # vanilla ref 'c:BIC' not present in this mod
				}
			}
			if = {
				limit = {
					NOT = {
						culture = {
							has_discrimination_trait_group = heritage_group_european
						}
					}
					OR = {
						is_pop_type = bureaucrats
						is_pop_type = clerks
					}
				}
				add = {
					desc = "POP_INDIA_NATIVE_ADMINISTRATORS"
					value = 25
				}
			}
			else_if = {
				limit = {
					NOT = {
						culture = {
							has_discrimination_trait_group = heritage_group_european
						}
					}
					NOR = {
						is_pop_type = bureaucrats
						is_pop_type = clerks
					}
				}
				multiply = {
					desc = "POP_INDIA_ADMINISTRATION_PRIVILEGES"
					value = 0.05
				}
			}
		}

		multiply = {
			desc = "LEADER_POPULARITY"
			scope:interest_group ?= {
				leader ?= {
					value = popularity
					multiply = 0.0025
					add = 1
					max = 1.25
					min = 0.75
				}
			}
		}

		multiply = {
			desc = "decentralized_power"
			value = 1.0

			if = {
				limit = {
					owner = {
						is_country_type = decentralized
					}
				}
				value = 0.0
			}
		}
		if = {
			limit = {
				scope:interest_group ?= {
					is_in_government = yes
				}
			}
			multiply = {
				desc = "IN_GOVERNMENT_ATTRACTION"
				value = 1
				add = scope:interest_group.modifier:interest_group_in_government_attraction_mult
				min = 0
			}
		}

		if = {
			limit = {
				scope:interest_group ?= {
					is_in_government = yes
				}
			}
			multiply = { # Multiplied by 1.25x max
				desc = "POP_LOYALISTS"
				value = pop_loyalist_fraction
				divide = 4
				add = 1
			}
		}
		if = {
			limit = {
				scope:interest_group ?= {
					is_in_government = no
				}
			}
			multiply = { # Multiplied by 1.25x max
				desc = "POP_RADICALS"
				value = pop_radical_fraction
				divide = 4
				add = 1
			}
		}
	}

	monarch_weight = {
		value = 1
		# Monarch is not likely to adopt a marginal IG
		if = {
			limit = {
				ig_counts_as_marginal = yes
			}
			multiply = {
				value = 0.1
			}
		}
		# Monarch is more likely to adopt a Powerful IG
		if = {
			limit = {
				is_powerful = yes
			}
			multiply = {
				value = 2
			}
		}
		# More likely with Wealth Voting or Oligarchy
		if = {
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_oligarchy
						has_law_or_variant = law_type:law_wealth_voting
					}
				}
			}
			multiply = {
				value = 2
			}
		}
		# Slightly more likely with LF
		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_laissez_faire
				}
			}
			multiply = {
				value = 1.25
			}
		}
		# Less likely before researching Mutual Funds
		if = {
			limit = {
				owner = {
					NOT = { has_technology_researched = mutual_funds }
				}
			}
			multiply = {
				value = 0.5
			}
		}
		# Impossible without Currency Standards
		if = {
			limit = {
				owner = {
					NOT = { has_technology_researched = transmutation_standards }
				}
			}
			multiply = {
				value = 0
			}
		}
	}

	agitator_weight = {
		# Agitators more likely to come from populist IG's rather than elitist
		value = 0.25
		# Agitators will never be part of a marginalized IG
		if = {
			limit = {
				owner.ig:ig_industrialists ?= {
					ig_counts_as_marginal = yes
				}
			}
			multiply = {
				value = 0
			}
		}
	}

		if = {

					has_law_or_variant = law_type:law_womens_suffrage
				}

			}
			add = {
				value = 0.025
			}
		}
		# Matriarchal branch: only women lead

			limit = {

					ova_country_is_matriarchal_society = yes

	}

	female_agitator_chance = {
		value = 0.025

		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_women_own_property
				}

			}
			add = {
				value = 0.025
			}
		}

		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_women_in_the_workplace
				}

			}
			add = {
				value = 0.025
			}
		}

		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_womens_suffrage
				}

			}
			add = {
				value = 0.025
			}
		}
		# Matriarchal branch: only women lead
		if = {
			limit = {
				owner = {
					ova_country_is_matriarchal_society = yes
				}
			}
			value = 0.975
		}

		if = {
			limit = {
				owner = {
					OR = {
						has_law = law_type:law_men_own_property
						has_law = law_type:law_men_in_the_workplace
						has_law = law_type:law_mens_suffrage
					}
				}
			}
			subtract = 0.025
		}
	}
