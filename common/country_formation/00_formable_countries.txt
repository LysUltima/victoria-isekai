# Standard culture formables (empire + kingdom tier alike). Each claims the former's
# primary-culture states via use_culture_states and requires holding 70% of them.
# None are major/unification formations -- that mechanic (unification_play, is_major_
# formation) is reserved for real historical unifications and is not used here.
# FRM (norse) and VMO (western-isles) are multi-culture realms: either culture may
# form them. Target defs: country_definitions/20_formables.txt.

#### Empire-tier formables ####

SAK = { # Sakuragoku (sakura)
	use_culture_states = yes
	required_states_fraction = 0.7

	possible = {
		country_has_primary_culture = cu:sakura
	}

	ai_will_do = { always = yes }
}

MAE = { # Prunhwa (mae)
	use_culture_states = yes
	required_states_fraction = 0.7

	possible = {
		country_has_primary_culture = cu:mae
        NOT = { country_has_state_religion = rel:diabolist }
	}

	ai_will_do = { always = yes }
}

VMO = { # Vaimotu (whenua + kanaka)
