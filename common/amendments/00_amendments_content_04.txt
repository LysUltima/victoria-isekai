amendment_reinstated_fueros = {
	parent = law_subjecthood

	allowed_laws = {
		law_subjecthood
	}

	modifier = {
		interest_group_ig_landowners_approval_add = 1
		interest_group_ig_rural_folk_approval_add = 1
		country_two_spains_conservative_drift_add = 0.1
	}

	possible = {
		always = no
	}

	would_sponsor = {
		AND = {
			exists = scope:approval
			scope:approval > 0
		}
	}

	ai_will_revoke = {
		always = no
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	on_activate = {
		if = {
			limit = {
				any_scope_state = {
					always = no # vanilla ref 's:STATE_BASQUE_COUNTRY' not present in this mod
				}
			}
			random_scope_state = {
				limit = {
					always = no # vanilla ref 's:STATE_BASQUE_COUNTRY' not present in this mod
				}
				add_modifier = spc_state_fueros_restored
			}
		}
		# removed effect line: was culture = cu:basque (no mod analogue)
	}

	on_deactivate = {
		if = {
			limit = {
				NAND = {
					always = no
					# removed: any_in_list on dead variable 'spain_accepted_cultures', never set elsewhere
					always = no
				}
			}
			# removed effect line: was culture = cu:basque (no mod analogue)
		}
		if = {
			limit = {
				any_scope_state = {
					always = no # vanilla ref 's:STATE_BASQUE_COUNTRY' not present in this mod
				}
			}
			random_scope_state = {
				limit = {
					always = no # vanilla ref 's:STATE_BASQUE_COUNTRY' not present in this mod
				}
				remove_modifier = spc_state_fueros_restored
			}
			add_radicals = {
				# removed effect line: was culture = cu:basque (no mod analogue)
				value = {
					add = huge_radicals
				}
			}
		}
	}

	amendment_activism_multiplier = 0.50
}
