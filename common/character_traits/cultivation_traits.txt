### ISEKAI TC: Cultivation realm traits
# Ladder of condition traits mirroring the Spiritualism cultivation tech line.
# Each realm requires the matching technology and the previous realm trait,
# and replaces the trait below it. Cultivation extends lifespan (health),
# sharpens battlefield command, and awes the masses (popularity).

trait_qi_gathering = {
	type = condition
	texture = "gfx/interface/icons/character_trait_icons/meticulous.dds"

	character_modifier = {
		character_health_add = 0.1
		character_popularity_add = 5
	}

	command_modifier = {
		unit_morale_recovery_mult = 0.05
	}

	country_modifier = {
	}

	agitator_modifier = {
	}

	interest_group_modifier = {
	}

	possible = {
		owner ?= { has_technology_researched = qi_gathering }
	}

	weight = {
		value = 20
		if = {
			limit = {
				owner ?= { has_technology_researched = foundation_establishment }
			}
			multiply = 1.5
		}
	}

	replace = {
	}
}
