AI = {
	# Only independent countries receive conquest goals. Subjects can still be targets.
	every_country = {
		limit = { is_subject = no }
		save_scope_as = conquering_country

		every_primary_culture = {
			limit = {
				OR = {
					this = cu:horsekin
					this = cu:bearkin
				}
			}

			every_country = {
				limit = {
					NOT = { this = scope:conquering_country }
					is_adjacent_to_country = scope:conquering_country
					country_has_primary_culture = prev
				}
				scope:conquering_country = {
					set_secret_goal = { country = prev secret_goal = conquer }
				}
			}
		}
	}
}
