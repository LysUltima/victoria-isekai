# Beastland companies.

# Catpital Granary Guild
company_ova_catfood_guild = {
	icon = "gfx/interface/icons/company_icons/cat2.dds"

	background = "gfx/interface/icons/company_icons/company_backgrounds/comp_illu_farm_wheat.dds"

	category = aristocrat_owned

	flavored_company = yes

	uses_dynamic_naming = no

	preferred_headquarters = { STATE_CATPITAL }

	building_types = { building_wheat_farm building_food_industry }

	extension_building_types = { building_livestock_ranch building_fishing_wharf building_chemical_plant }

	potential = {
		has_interest_marker_in_region = sr:region_beastland_central
	}

	attainable = {
		hidden_trigger = {
			any_scope_state = { state_region = s:STATE_CATPITAL }
		}
	}

	possible = {
		has_technology_researched = spirit_fertilizer_refinement
		has_technology_researched = canneries
		any_scope_state = {
			state_region = s:STATE_CATPITAL
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_wheat_farm
				level >= 5
			}
		}
	}

	possible_prestige_goods = { prestige_good_ova_catpital_feast_baskets }

	prestige_goods_trigger = {
		custom_tooltip = {
			text = company_ova_catfood_guild_prestige_tt
			owner = {
				OR = {
					any_scope_state = {
						any_scope_building = {
							is_building_type = building_food_industry
							has_active_production_method = pm_bakery
						}
					}
					any_scope_state = {
						any_scope_building = {
							is_building_type = building_food_industry
							has_active_production_method = pm_cannery
						}
					}
				}
			}
		}
	}

	prosperity_modifier = {
		building_food_industry_throughput_add = 0.1
		state_radicalism_increases_violent_hostility_mult = -0.05
		state_radicalism_increases_cultural_erasure_mult = -0.05
		state_radicalism_increases_open_prejudice_mult = -0.05
	}

	ai_will_do = {
		has_technology_researched = spirit_fertilizer_refinement
		has_technology_researched = canneries
		any_scope_state = {
			state_region = s:STATE_CATPITAL
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_wheat_farm
				level >= 5
			}
		}
	}

	ai_construction_targets = {
		building_wheat_farm = {
			level = 5
			state_trigger = {
				is_incorporated = yes
				state_region = s:STATE_CATPITAL
			}
		}
	}

	ai_weight = { value = 3 }
}
