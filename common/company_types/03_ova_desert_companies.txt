	prestige_goods_trigger = {

							has_active_production_method = pm_sparse_planting_building_coffee_plantation
						}
					}
					any_scope_state = {
						any_scope_building = {
							is_building_type = building_coffee_plantation

			}
		}
	}

	prosperity_modifier = {

	ai_will_do = {

			level = 5
