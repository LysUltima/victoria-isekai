	prestige_goods_trigger = {
		custom_tooltip = {
			text = company_ova_adventurers_guild_prestige_tt

				OR = {
					any_scope_state = {
						any_scope_building = {
							is_building_type = building_spirit_beast_hunting
							has_active_production_method = pm_spirit_beast_hunting
						}
					}
					any_scope_state = {
						any_scope_building = {
							is_building_type = building_spirit_beast_hunting
							has_active_production_method = pm_warded_hunting_parties

	prosperity_modifier = {

				is_incorporated = yes
				any_scope_building = {
					is_building_type = building_whaling_station
					level >= 3
				}
			}
		}
	}

	ai_construction_targets = {
		building_spirit_beast_hunting = {
			level = 3
			state_trigger = {
				is_incorporated = yes
			}
		}
		building_whaling_station = {
			level = 3
			state_trigger = {
				is_incorporated = yes
			}
