### Isekai hero storyline - journal entries
# Victoria's arc is a chain of NINE sequential journal entries
# (je_ova_victoria_1 .. je_ova_victoria_9), each with its own objective and
# each added by the previous one's on_complete. JEs 2-9 define no
# is_shown_when_inactive (default no) so they only ever enter via the chain -
# same precedent as the murim returner chain. Chapter 9 is the LAST entry:
# completing it fires the choice event (ova_hero.75) sixty days later, with
# no further journal entry in between. All chain events roll on the YEARLY
# pulse (user direction, 2026-07-15).
# The rebellion path (chosen in ova_hero.75) sets the global variable
# ova_truth_exposed (a fail condition for the crusade and every
# summoning-pact JE in the world), splits the demonland marches off as AUR,
# the Concord Aurorae, under Victoria, and starts the winner-takes-all war
# of the dawn between AUR and the old crown (which keeps the tag LYS). The
# three je_ova_dawn_* aftermath entries run on AUR.

je_ova_eternal_crusade = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_foreign_affairs

	on_monthly_pulse = {
		effect = {
			# recount living OTHERWORLDER heroes only (trait_otherworlder, not
			# every mundane champion) - prestige scales with that count
			set_variable = { name = ova_lux_hero_count value = 0 }
			every_country = {
				limit = {
					has_variable = ova_champion_char
					exists = var:ova_champion_char
					var:ova_champion_char ?= { has_trait = trait_otherworlder }
				}
				c:LUX ?= {
					change_variable = { name = ova_lux_hero_count add = 1 }
				}
			}
			if = {
				limit = { has_modifier = ova_lux_summoned_legion }
				remove_modifier = ova_lux_summoned_legion
			}
			if = {
				limit = { var:ova_lux_hero_count >= 1 }
				add_modifier = {
					name = ova_lux_summoned_legion
					days = -1
					multiplier = var:ova_lux_hero_count
				}
			}
		}
		random_events = {
			5000 = 0
			30 = ova_hero.41
			25 = ova_hero.42
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ova_eternal_crusade_status_active
				trigger = {
					var:ova_lux_hero_count >= 1
				}
			}
			triggered_desc = {
				desc = je_ova_eternal_crusade_status_none
				trigger = {
					always = yes
				}
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_ova_eternal_crusade_complete_tt
			NOT = { exists = c:DEM }
		}
	}

	on_complete = {
		if = {
			limit = { is_power_bloc_leader = yes }
			power_bloc = { remove_modifier = ova_unbreakable_covenant }
		}
		if = {
			limit = { has_modifier = ova_lux_summoned_legion }
			remove_modifier = ova_lux_summoned_legion
		}
		add_modifier = { name = ova_lux_crusade_triumphant days = -1 }
	}

	# The crusade dies as a cause if the world recognizes the Demonic Empire,
	# or if the Church stands discredited before the world (the rebellion path
	# of Victoria's arc exposes what the archives held).
	fail = {
		custom_tooltip = {
			text = je_ova_eternal_crusade_fail_tt
			OR = {
				AND = {
					exists = c:DEM
					c:DEM ?= { is_country_type = recognized }
				}
				has_global_variable = ova_truth_exposed
			}
		}
	}

	on_fail = {
		if = {
			limit = { is_power_bloc_leader = yes }
			power_bloc = { remove_modifier = ova_unbreakable_covenant }
		}
		if = {
			limit = { has_modifier = ova_lux_summoned_legion }
			remove_modifier = ova_lux_summoned_legion
		}
		add_modifier = { name = ova_lux_crusade_broken days = 3650 }
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
}

			}

	}

	weight = 900

	transferable = no
}

# ------------------------------------------------------------------------
# Victoria's arc, chapter 1 of 9: settle in, and build her a home.
# Objective: a Champion Hall stands with all three support wings staffed
# (every side PMG off its base "no program" PM).
# ------------------------------------------------------------------------
je_ova_victoria_1 = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_internal_affairs

	# Raised from the summoning flow, not self-activated. je_ova_hero_summoning
	# (itself raised from history) creates Victoria through a DEFERRED
	# create_character and fires ova_hero.2 three days on; that event adds this
	# chapter once she is confirmed alive (ova_champion_char / trait_otherworlder).
	# A one-time day-one history add would run this immediate before the deferred
	# character resolves, so the arc is deliberately hooked to ova_hero.2 instead.
	immediate = {
		set_variable = ova_victoria_arc_started
		set_variable = { name = ova_victoria_char value = var:ova_champion_char }
		# plot armor: the story needs her alive until it is told
		effect_ova_victoria_armor_on = yes
		trigger_event = { id = ova_hero.50 days = 10 }
	}

	# Three beats a year: six beats must land inside ~2 years, so the pulse
	# offers one now and schedules the relay event (ova_hero.110) twice, at
	# 120 and 240 days. See the relay's note in ova_hero_events.txt for why
	# the off-pulse beats ride an event.
	on_yearly_pulse = {
		effect = {
			effect_ova_victoria_ch1_tick = yes
			trigger_event = { id = ova_hero.110 days = 120 }
			trigger_event = { id = ova_hero.110 days = 240 }
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ova_victoria_1_status
				trigger = { always = yes }
			}
		}
	}

	# has_variable = ova_victoria_seen_50 (set by the chapter's own intro
	# event) keeps this from completing on day one: building_champion_hall is
	# a generic building, so a nation that already had a fully-staffed hall
	# (built for an earlier champion, say) would otherwise satisfy
	# ova_victoria_hall_ready the instant this chapter unlocked.
	complete = {
		custom_tooltip = {
			text = je_ova_victoria_1_complete_tt
			AND = {
				has_variable = ova_victoria_seen_50
				ova_victoria_hall_ready = yes
			}
		}
	}

	on_complete = {
		add_journal_entry = { type = je_ova_victoria_2 }
	}

	fail = {
		custom_tooltip = {
			text = je_ova_victoria_fail_tt
			OR = {
				AND = {
					has_variable = ova_victoria_char
					NOT = { exists = var:ova_victoria_char }
				}
				NOT = { ova_hero_crusade_active = yes }
			}
		}
	}

	on_fail = {
		effect_ova_victoria_story_over = yes
		trigger_event = { id = ova_hero.78 days = 3 }
	}

	weight = 800
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
}
