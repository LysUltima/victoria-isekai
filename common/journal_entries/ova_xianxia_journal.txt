### Xianxia storyline - journal entries
# Blood Lotus Covenant (c:BLO leads GHO/SHA/NIN) vs Great Feng (FEN) + Great Yao (YAO) and their
# sworn sects, plus the Destined One chain. Per tools/design/XIANXIA_STORYLINE_SPEC.md: NONE of
# these JEs call ova_champion_monthly_tick (the universal je_ova_champion owns that) and none use
# a current_value champion-level progress bar - the champion foundation's own two scripted
# progress bars already show level/renown everywhere. The day-one cohesion lock on BLO's Blood
# Lotus Covenant bloc is applied from history (common/history/power_blocs/ova_power_blocs.txt),
# not from any JE immediate; only the exit paths below remove it.
#
# 2026-07-18 rework (Victoria-chain precedent, user-directed):
#  - The four war JEs are STARTED FROM HISTORY (common/history/global/00_global.txt adds the
#    journal entries and fires the intro proclamations ova_xianxia.3/.4/.5). They define no
#    is_shown_when_inactive, no possible, and no immediate - history owns day one.
#  - The Destined One chain is now SIX chapters, one per cultivation realm
#    (je_ova_xianxia_mc_1 Qi Gathering .. je_ova_xianxia_mc_6 Return to the Void), each added by
#    the previous chapter's on_complete. Every chapter is a strict beat chain dispatched through
#    an if/else_if ladder (effect_ova_xianxia_chN_tick) from on_yearly_pulse plus hidden relay
#    events (ova_xianxia.130-.135), NOT through random_events - a uniform roll would land on
#    beats whose turn has passed and a five-beat chapter would crawl for a decade. Cadence rule
#    (from the Victoria chain): beats / per-year rate <= ~2 years per chapter.
#  - Every chapter has MULTIPLE goals: its own tale told (the final beat's seen flag, which the
#    ladder alone can set - the Victoria seen-flag guard, so no chapter completes with zero
#    beats shown), the next realm's level threshold, and one or more jianghu deeds driven by
#    decisions (secret manual, secret realm, rival duel or grand competition, natal weapon,
#    tribulation treasures). The archetype cheats thread through dedicated ladder rungs in
#    chapters 1/3/5 (.25-.29 / .20-.24 / .85-.89) picked by ova_xianxia_mc_type.

# ============================================================================
# The Destined One chain - six realms, strictly sequential.
# ============================================================================
# All six are host-country only: they only ever activate via add_journal_entry
# (effect_ova_xianxia_mc_arrives / on_complete chaining). Like the foundation's
# je_ova_champion_coup they define NO is_shown_when_inactive (default no) - an
# is_shown_when_inactive keyed on the host variable would self-activate later
# realms the moment the Destined One arrives, skipping the chain. Shared
# death-fail (repeated verbatim per Paradox's non-reusable JE-body syntax):
# the Destined One dead -> effect_ova_xianxia_story_over (halo off) + the
# funeral event .72. The realm status_desc chain is likewise repeated verbatim
# in every chapter.

# ------------------------------------------------------------------------
# Realm 1 of 6: Qi Gathering. A nobody in plain robes sweeps the ten
# thousand steps and learns to hold one breath of qi.
# Beats: .100 -> .16 -> the gift stirs (.25-.29 by archetype) -> .17 -> .18
# (ladder in effect_ova_xianxia_ch1_tick; relay ova_xianxia.130 at 120/240
# days = 3 beats a year, 5 beats in ~1.7 years).
# Goals: the tale told (seen_18), Foundation Establishment reached (level
# 10), and a first sliver of renown.
# ------------------------------------------------------------------------
je_ova_xianxia_mc_1 = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_internal_affairs

	immediate = {
		add_modifier = { name = ova_xianxia_destined_halo days = -1 }
		set_variable = { name = ova_xianxia_rivals_defeated value = 0 }
		trigger_event = { id = ova_xianxia.100 days = 10 }
	}

	# The lazy character capture must run soon after arrival, so it keeps the
	# monthly pulse; the story beats ride the yearly tick ladder below.
	on_monthly_pulse = {
		effect = {
			# create_character is deferred - the arrival effect could not save the Destined One
			# directly, so the first pulse after its champion record exists captures the character
			# into the mc-specific variable.
			if = {
				limit = {
					NOT = { has_variable = ova_xianxia_mc_char }
					has_variable = ova_champion_char
					exists = var:ova_champion_char
				}
				set_variable = { name = ova_xianxia_mc_char value = var:ova_champion_char }
			}
		}
	}

	# Three beats a year: the pulse offers one now and schedules the relay
	# event (ova_xianxia.130) twice, at 120 and 240 days.
	on_yearly_pulse = {
		effect = {
			effect_ova_xianxia_ch1_tick = yes
			trigger_event = { id = ova_xianxia.130 days = 120 }
			trigger_event = { id = ova_xianxia.130 days = 240 }
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ova_xianxia_realm_none
				trigger = {
					NOT = { ova_has_living_champion = yes }
				}
			}
			triggered_desc = {
				desc = je_ova_xianxia_realm_qi_gathering
				trigger = {
					var:ova_champion_level < 10
				}
			}
			triggered_desc = {
				desc = je_ova_xianxia_realm_foundation
				trigger = {
					var:ova_champion_level < 20
				}
			}
			triggered_desc = {
				desc = je_ova_xianxia_realm_golden_core
				trigger = {
					var:ova_champion_level < 30
				}
			}
			triggered_desc = {
				desc = je_ova_xianxia_realm_nascent_soul
				trigger = {
					var:ova_champion_level < 40
				}
			}
			triggered_desc = {
				desc = je_ova_xianxia_realm_soul_formation
				trigger = {
					var:ova_champion_level < 50
				}
			}
			triggered_desc = {
				desc = je_ova_xianxia_realm_return_void
				trigger = {
					always = yes
				}
			}
		}
	}

	# seen_18 is only ever set by this chapter's own final beat, so a youth
	# who leveled fast cannot skip the story (the Victoria seen-flag guard).
	complete = {
		custom_tooltip = {
			text = je_ova_xianxia_mc_1_tale_tt
			has_variable = ova_xianxia_seen_18
		}
		custom_tooltip = {
			text = je_ova_xianxia_mc_1_level_tt
			ova_has_living_champion = yes
			var:ova_champion_level >= 10
		}
		custom_tooltip = {
			text = je_ova_xianxia_mc_1_name_tt
			ova_has_living_champion = yes
			var:ova_champion_popularity >= 1
		}
	}

	on_complete = {
		trigger_event = { id = ova_xianxia.63 days = 3 }
		add_journal_entry = { type = je_ova_xianxia_mc_2 }
	}

	fail = {
		custom_tooltip = {
			text = je_ova_xianxia_mc_dead_tt
			has_variable = ova_xianxia_mc_char
			NOT = { exists = var:ova_xianxia_mc_char }
		}
	}

	on_fail = {
		effect_ova_xianxia_story_over = yes
		trigger_event = { id = ova_xianxia.72 days = 3 }
	}

	weight = 4000
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
}
