### Murim storyline - journal entries
# Heavenly Demon Sect (c:CHE) vs Orthodox Martial Alliance, plus the Returner (Kang Jin-woo)
# chain. NONE of these JEs call ova_champion_monthly_tick (the universal je_ova_champion owns
# that) and none use a current_value champion-level progress bar - the champion foundation's
# own two scripted progress bars already show level/renown everywhere. The day-one cohesion
# lock on HWA's Orthodox Martial Alliance bloc is applied from history
# (common/history/power_blocs/ova_power_blocs.txt), not from any JE immediate; only the exit
# paths below remove it.
#
# 2026-07-18 second rework (the hero's road). The chain:
#   returner_1 (the strange disciple's rise, 8 beats, level 20)
#   -> je_ova_murim_ally_home (his OWN banner first: staff a Champion Hall, win the young
#      mistress/senior of his house) -> up to 8 more je_ova_murim_ally_* courtships, each a
#      4-beat chain (intro creates the companion, two TRIAL beats with a selfless and a
#      selfish option, the pledge) + a clearly stated objective (armies, buildings, realm
#      ranks, decisions, relations) under a hard FIVE-YEAR timeout
#   -> the succession endgame (.150: demonic cult assassins kill the old master; the
#      Returner takes the seat only now, at the very last moment)
#      GOOD (>= 75% selfless choices AND his own house courted without one selfish choice):
#        wedding into the host clan / acclamation as sect patriarch, every sworn banner
#        becomes a MARCH, then je_ova_murim_final_battle (a champion expedition into the
#        Cult triggers the duel he wins)
#      BAD: the seat taken by force (plot armor off), je_ova_murim_war_conference (raise
#        30 battalions), then the scripted Reckoning war (dp_ova_murim_reckoning, mirrored
#        full-annexation goals)
#   Both endings run effect_ova_murim_shatter_cult: the Cult's five mountains become the
#   five successor sects (marches of the host in the good ending), conquered banner states
#   go home, and whatever the Cult held outside the murim falls to the host.
#
# Beats are dispatched through if/else_if tick ladders (effect_ova_murim_*_tick) from
# on_yearly_pulse plus hidden relay events (ova_murim.90/.93/.94), NOT through
# random_events - a uniform roll would land on beats whose turn has passed. Cadence rule
# (from the Victoria chain): beats / per-year rate <= ~2 years per chapter.

# ============================================================================
# The Returner chain (Kang Jin-woo) - strictly sequential: the rise, then the
# Eight Banners arc (his own house first), then the succession endgame and one
# of two endings. All chain JEs are host-country only: they only ever activate
# via add_journal_entry (effect_ova_murim_returner_arrives / on_complete /
# effect_ova_murim_next_ally chaining). Like the foundation's
# je_ova_champion_coup they define NO is_shown_when_inactive (default no) - an
# is_shown_when_inactive keyed on the host variable would self-activate later
# chapters the moment the Returner arrives, skipping the chain. Shared fail
# (repeated verbatim per Paradox's non-reusable JE-body syntax): the Returner
# dead, OR the Cult has mastered the murim - on_fail routes through
# effect_ova_murim_fail_events (.72 death / .73 calamity) and
# effect_ova_murim_story_over (insight off, plot armor off). The realm
# status_desc chain (Third-Rate .. Life-and-Death Realm) is likewise repeated
# verbatim in every chapter.
# ============================================================================

# ------------------------------------------------------------------------
# Chapter 1: The Strange Disciple. He kneels at the gate, and spends two
# years being right about things a beggar boy has no business being right
# about. Beats: .3 -> .10 -> .11 -> .16 -> .12 -> .13 -> .15 -> .14 (ladder
# in effect_ova_murim_ch1_tick; relay ova_murim.90 at 90/180/270 days = 4
# beats a year, 8 beats in ~2 years). Objective: the chapter's beats told,
# and the Returner risen to the First-Rate rank (level 20).
# ------------------------------------------------------------------------
je_ova_murim_returner_1 = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_internal_affairs

	immediate = {
		add_modifier = { name = ova_murim_returners_insight days = -1 }
	}

	# The lazy character capture must run soon after arrival, so it keeps the
	# monthly pulse; the story beats ride the yearly tick ladder below.
	on_monthly_pulse = {
		effect = {
			# create_character is deferred - the arrival effect could not save
			# him directly, so the first pulse after his champion record exists
			# captures the character into the Returner-specific variable.
			if = {
				limit = {
					NOT = { has_variable = ova_murim_returner_char }
					has_variable = ova_champion_char
					exists = var:ova_champion_char
				}
				set_variable = { name = ova_murim_returner_char value = var:ova_champion_char }
				# plot armor: the Returner's story needs him alive until his
				# tale resolves (the good ending keeps it to the epilogue; the
				# seat taken by force strips it early)
				var:ova_champion_char ?= {
					set_variable = ova_plot_armor
					set_character_immortal = yes
				}
			}
		}
	}

	# Four beats a year: the pulse offers one now and schedules the relay
	# event (ova_murim.90) three times, at 90, 180 and 270 days.
	on_yearly_pulse = {
		effect = {
			effect_ova_murim_ch1_tick = yes
			trigger_event = { id = ova_murim.90 days = 90 }
			trigger_event = { id = ova_murim.90 days = 180 }
			trigger_event = { id = ova_murim.90 days = 270 }
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ova_murim_realm_none
				trigger = {
					NOT = { ova_has_living_champion = yes }
				}
			}
			triggered_desc = {
				desc = je_ova_murim_realm_third_rate
				trigger = {
					var:ova_champion_level < 10
				}
			}
			triggered_desc = {
				desc = je_ova_murim_realm_second_rate
				trigger = {
					var:ova_champion_level < 20
				}
			}
			triggered_desc = {
				desc = je_ova_murim_realm_first_rate
				trigger = {
					var:ova_champion_level < 30
				}
			}
			triggered_desc = {
				desc = je_ova_murim_realm_peak
				trigger = {
					var:ova_champion_level < 40
				}
			}
			triggered_desc = {
				desc = je_ova_murim_realm_profound
				trigger = {
					var:ova_champion_level < 50
				}
			}
			triggered_desc = {
				desc = je_ova_murim_realm_life_death
				trigger = {
					always = yes
				}
			}
		}
	}

	# ova_murim_seen_14 is only ever set by this chapter's own final beat, so
	# a champion who leveled fast cannot skip the story (the Victoria chain's
	# seen-flag guard, applied from the start here).
	complete = {
		custom_tooltip = {
			text = je_ova_murim_returner_1_complete_tt
			AND = {
				has_variable = ova_murim_seen_14
				ova_has_living_champion = yes
				var:ova_champion_level >= 20
			}
		}
	}

	on_complete = {
		# seven gates and his own besides - the Eight Banners arc begins with
		# the house that took him in
		effect_ova_murim_next_ally = yes
	}

	fail = {
		custom_tooltip = {
			text = je_ova_murim_returner_fail_tt
			OR = {
				AND = {
					has_variable = ova_murim_returner_char
					NOT = { exists = var:ova_murim_returner_char }
				}
				ova_murim_cult_has_won_trigger = yes
			}
		}
	}

	on_fail = {
		effect_ova_murim_story_over = yes
		effect_ova_murim_fail_events = yes
	}

	weight = 4000
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
}
