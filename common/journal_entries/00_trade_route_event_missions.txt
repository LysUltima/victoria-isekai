je_build_local_arms_industry = {
	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_internal_affairs

	timeout = 1827

	weight = 200

	transferable = no

	should_be_pinned_by_default_uninvolved_or_context = yes

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_state = {
				has_active_building = building_arms_industry
			}
		}
		effect = {
			# removed: vanilla ref 'modifier_reliable_supply' (defined only in blanked montenegrin_modifiers.txt) not present in this mod
			ig:ig_armed_forces ?= {
				add_modifier = {
					name = ig_domestic_weapons_manufacturing
					days = long_modifier_time
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		flags = { timeout  }
		conclusion_desc = je_standard_failure_conclusion_desc
		effect = {
			ig:ig_armed_forces ?= {
				add_modifier = {
					name = ig_unreliable_weapon_imports
					days = long_modifier_time
					multiplier = 2
				}
			}
		}
	}
}

je_set_up_grain_import = {
	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_foreign_affairs

	timeout = 730

	weight = 200

	transferable = no

	should_be_pinned_by_default_uninvolved_or_context = yes

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			capital = {
				sg:grain = {
					or = {
						import_tariff_level = low_subventions
						import_tariff_level = high_subventions
						import_tariff_level = max_subventions
					}
				}
			}
			capital.market = {
				NOT = {
					mg:grain = { market_goods_pricier > 0.20 }
				}
			}
		}
		effect = {
			scope:target = {
				add_modifier = {
					name = ig_secured_desired_imports
					days = normal_modifier_time
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		flags = { timeout  }
		conclusion_desc = je_standard_failure_conclusion_desc
		effect = {
			scope:target = {
				add_modifier = {
					name = ig_refused_desired_imports
					days = normal_modifier_time
				}
			}
		}
	}
}
