### Saint Christine's Academy - a FOUR-YEAR school for summoned Heroes.
# Split out of ova_hero_events.txt (2026-07-23) into its own namespace.
#
# Structure (Korean academy-novel beats):
#   ova_academy.1        enrollment intro (fired from ova_hero_academy_decision)
#   ova_academy.10       the TERM dispatcher (hidden; counter-routed)
#   ova_academy.20       the GRADUATION dispatcher (hidden)
#   ova_academy.101-.104 Term 1 pool (Year 1, Semester 1 - arrival/awakening)
#   ova_academy.111-.114 Term 2 pool (Year 1, Semester 2 - first finals)
#   ova_academy.121-.124 Term 3 pool (Year 2, Semester 1 - practicals begin)
#   ova_academy.131-.134 Term 4 pool (Year 2, Semester 2 - festival + terror)
#   ova_academy.141-.144 Term 5 pool (Year 3, Semester 1 - scouted/tower)
#   ova_academy.151-.154 Term 6 pool (Year 3, Semester 2 - trials/rival-ally)
#   ova_academy.161-.164 Term 7 pool (Year 4, Semester 1 - senior/tournament)
#   ova_academy.171-.174 Term 8 pool (Year 4, Semester 2 - run-up to graduation)
#   ova_academy.181-.184 Graduation pool (the four graduation scenes)
#
# The course runs 1460 days. TWO term-beats fire per 365-day year (eight in
# total), then graduation. Each term draws at random from its OWN unique pool
# of four - no pool is ever reused for another term. A running counter
# (ova_academy_term) tells the dispatcher which term it is.
#
# VICTORIA (c:LYS) keeps her four private chapter-2 beats (ova_hero.92-.95),
# and they fire AT THE APPROPRIATE TIME: one per year, on the second term of
# each year, so .92 (year one) ... .95 (year four, the Final Practicum) lands
# as the very last beat before graduation. On her other four terms she draws
# that term's generic pool like anyone else. Any non-Victoria Hero simply walks
# all eight generic pools in order.
#
# The graduation variable is kept under its old name (ova_hero_academy_graduated)
# because the decision gate, je_ova_victoria_2's complete, ova_hero.35's trigger
# and the champion-death cleanup all still read it.
# All champion XP/popularity effects are the foundation's (ova_champion_*).

namespace = ova_academy

# ------------------------------------------------------- .10 the term dispatcher
# Hidden - it only routes. Increments the term counter, weaves in Victoria's
# private year-beat where one is due, otherwise fires this term's own pool.
ova_academy.10 = {
	type = country_event
	hidden = yes

	trigger = {
		ova_has_living_champion = yes
	}

	immediate = {
		change_variable = { name = ova_academy_term add = 1 }

		# Victoria's four private beats: one per year, on the second term of
		# each year (terms 2/4/6/8), .92 -> .95 in order. Each falls through to
		# the generic pool if she has somehow already seen it.
		if = {
			limit = {
				c:LYS ?= THIS
				var:ova_academy_term = 2
				NOT = { has_variable = ova_victoria_seen_92 }
			}
			trigger_event = { id = ova_hero.92 }
		}
		else_if = {
			limit = {
				c:LYS ?= THIS
				var:ova_academy_term = 4
				NOT = { has_variable = ova_victoria_seen_93 }
			}
			trigger_event = { id = ova_hero.93 }
		}
		else_if = {
			limit = {
				c:LYS ?= THIS
				var:ova_academy_term = 6
				NOT = { has_variable = ova_victoria_seen_94 }
			}
			trigger_event = { id = ova_hero.94 }
		}
		else_if = {
			limit = {
				c:LYS ?= THIS
				var:ova_academy_term = 8
				NOT = { has_variable = ova_victoria_seen_95 }
			}
			trigger_event = { id = ova_hero.95 }
		}
		# generic pools, one unique set of four per term
		else_if = {
			limit = { var:ova_academy_term = 1 }
			random_list = {
				25 = { trigger_event = { id = ova_academy.101 } }
				25 = { trigger_event = { id = ova_academy.102 } }

			random_list = {
				25 = { trigger_event = { id = ova_academy.131 } }
				25 = { trigger_event = { id = ova_academy.132 } }
				25 = { trigger_event = { id = ova_academy.133 } }
				25 = { trigger_event = { id = ova_academy.134 } }
			}
		}

				25 = { trigger_event = { id = ova_academy.142 } }
