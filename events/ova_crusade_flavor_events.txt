### Generic crusade flavor events - the far war as the whole world sees it
# Namespace ova_crusade_flavor. Fired from the regular on_action
# crusade_flavor_yearly_events (common/on_actions/ova_flavor_on_actions.txt),
# hooked into the engine's on_yearly_pulse_country
# (common/on_actions/00_code_on_actions.txt) - do not renumber these fixed ids,
# the on_action's weights are wired against them.
#
# DELIBERATELY SURFACE-LEVEL. These events repeat the Church's public story of
# the Eternal Crusade at face value: a holy war against the Demon Lord, far
# away, glorious and profitable and slightly absurd. They fire for ANY
# centralized nation anywhere in the world (only DEM and LUX themselves are
# excluded - both have their own event sets) and must NEVER touch the truth:
# no angels, no doubted scriptures, no sympathy for the demons' claim. The slow
# reveal belongs to the Victoria chain and the DEM home front alone.
#
#  .1  penny serials about the summoned Heroes
#  .2  an alms collection for the holy war
#  .3  a mercenary comes home rich from the front
#  .4  a sideshow exhibits a "captured demon"
#  .5  a summoned Hero passes through on the way to the front
#  .6  crusade quartermasters drive prices up
#  .7  local clergy debate someone else's holy war
#  .8  local youths run off to join the crusade
#  .9  a supply contract for the crusade
#  .10 two broadsheets, two opposite battle reports

namespace = ova_crusade_flavor

ova_crusade_flavor.1 = { # the crusade in cheap print
	type = country_event
	placement = root
	duration = 3

	title = ova_crusade_flavor.1.t
	desc = ova_crusade_flavor.1.d
	flavor = ova_crusade_flavor.1.f

	event_image = { video = "unspecific_world_fair" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		ova_hero_crusade_active = yes
		NOT = { is_country_type = decentralized }
		NOT = { c:DEM ?= THIS }
		NOT = { c:LUX ?= THIS }
	}

	immediate = { }

	option = { # let the presses run
		name = ova_crusade_flavor.1.a
		default_option = yes
		add_loyalists = { value = very_small_radicals }
	}
	option = { # confiscate the trash
		name = ova_crusade_flavor.1.b
		add_radicals = { value = very_small_radicals }
	}
}
