### Champions JE - generic FLAVOR events (all countries with a champion)
# Namespace ova_champion_flavor. Fired from je_ova_champion's on_monthly_pulse
# random_events (the one foundation edit the flavor agent is allowed) and from
# the coup journal entry (je_ova_champion_coup). Uses ONLY the shared champion
# interface plus verified vanilla effects and the flavor helpers in
# ova_champion_flavor_effects.txt.
#
#  .1 -.6   Champion DEMANDS. What the champion pushes for matches their own
#           ideology / interest group (like the demands interest groups make
#           over a law). Granting starts the reform through the NORMAL
#           legislative process (start_enactment - not instant) and builds
#           renown; a lesser gesture placates them cheaply; refusing stirs the
#           matching movement and risks a coup.
#  .12      The overmighty-champion stir (very high renown), can lead to a coup.
#  .30-.36  Minor renown / XP flavor.
#  .40-.49  Ten more flavor events, inconsequential to consequential.
#  .60-.62  Coup pulse events (player agency while je_ova_champion_coup runs).
#  .63/.64  Coup resolution (throne seized / coup broken and champion exiled).
#
# Coups and movement stirs are REPEATABLE - no once-only guard.

namespace = ova_champion_flavor

# =========================================================================
#  DEMANDS. Common demand gate (shared by .1-.6): champion available, at home,
#  not the ruler, no coup running, no law already enacting, and renown high
#  enough that the crown must take them seriously.
# =========================================================================

ova_champion_flavor.1 = { # the Champion's audience: a negotiation (see iberia negotiation.1)
	type = country_event
	placement = root
	is_popup = yes
	duration = 3

	title = ova_champion_flavor.1.t

	desc = {
		first_valid = {
			triggered_desc = { desc = ova_champion_flavor.1.d  trigger = { var:ova_demand_variant = 1 } }
			triggered_desc = { desc = ova_champion_flavor.1.d2 trigger = { var:ova_demand_variant = 2 } }
			triggered_desc = { desc = ova_champion_flavor.1.d3 trigger = { always = yes } }
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = { desc = ova_champion_flavor.1.f  trigger = { var:ova_demand_variant = 1 } }
			triggered_desc = { desc = ova_champion_flavor.1.f2 trigger = { var:ova_demand_variant = 2 } }
			triggered_desc = { desc = ova_champion_flavor.1.f3 trigger = { always = yes } }
		}
	}

	event_image = { trigger = { var:ova_demand_variant = 1 } video = "unspecific_politicians_arguing" }
	event_image = { trigger = { var:ova_demand_variant = 2 } video = "unspecific_signed_contract" }
	event_image = { trigger = { var:ova_demand_variant = 3 } video = "unspecific_ruler_speaking_to_people" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		ova_champion_is_available = yes
		NOT = { ova_champion_is_adventuring = yes }
		NOT = { has_journal_entry = je_ova_champion_coup }
		NOT = { exists = currently_enacting_law }
		var:ova_champion_popularity >= 30
		var:ova_champion_char ?= { is_ruler_of_own_country = no }
	}

	immediate = {
		set_variable = ova_champion_demand_pending
		ova_champion_set_demand_options = yes
	}

	# whatever the crown decides, tidy the per-demand option variables
	after = {
		ova_champion_clear_demand_options = yes
		remove_variable = ova_champion_demand_pending
	}

	option = { # 1 - buy them off (bribe)
		name = ova_champion_flavor.1.o1
		trigger = { OR = { var:ova_demand_opt_a = 1 var:ova_demand_opt_b = 1 var:ova_demand_opt_c = 1 } }
		ai_chance = { base = 10 }
		ova_champion_bribe_demand = yes
	}
	option = { # 2 - a seat at the council (high office)
		name = ova_champion_flavor.1.o2
		trigger = { OR = { var:ova_demand_opt_a = 2 var:ova_demand_opt_b = 2 var:ova_demand_opt_c = 2 } }
		ai_chance = { base = 10 }
		ova_champion_add_popularity = { AMOUNT = 10 }
		add_modifier = { name = ova_champion_high_office days = 730 }
	}
	option = { # 3 - pledge to grow the army
		name = ova_champion_flavor.1.o3
		trigger = { OR = { var:ova_demand_opt_a = 3 var:ova_demand_opt_b = 3 var:ova_demand_opt_c = 3 } }
		ai_chance = { base = 10 }
		add_treasury = -30000
		ova_champion_add_popularity = { AMOUNT = 8 }
		add_modifier = { name = ova_champion_army_pledge days = 730 }
	}
	option = { # 4 - champion their cause (start the reform)
		name = ova_champion_flavor.1.o4
		trigger = {
			ova_champion_demand_law_available = yes
			OR = { var:ova_demand_opt_a = 4 var:ova_demand_opt_b = 4 var:ova_demand_opt_c = 4 }
		}
		ai_chance = { base = 10 }
		ova_champion_grant_cause_law = yes
	}
	option = { # 5 - fund their grand works
		name = ova_champion_flavor.1.o5
		trigger = { OR = { var:ova_demand_opt_a = 5 var:ova_demand_opt_b = 5 var:ova_demand_opt_c = 5 } }
		ai_chance = { base = 5 }
		add_treasury = -50000
		ova_champion_add_popularity = { AMOUNT = 10 }
		add_modifier = { name = ova_champion_patronage days = 365 }
	}
	option = { # 6 - cut taxes to please the people
		name = ova_champion_flavor.1.o6
		trigger = {
			tax_level >= medium
			OR = { var:ova_demand_opt_a = 6 var:ova_demand_opt_b = 6 var:ova_demand_opt_c = 6 }
		}
		ai_chance = { base = 5 }
		set_tax_level = low
		ova_champion_add_popularity = { AMOUNT = 8 }
	}
	option = { # 7 - lavish largesse on the people
		name = ova_champion_flavor.1.o7
		trigger = { OR = { var:ova_demand_opt_a = 7 var:ova_demand_opt_b = 7 var:ova_demand_opt_c = 7 } }
		ai_chance = { base = 5 }
		add_treasury = -40000
		ova_champion_add_popularity = { AMOUNT = 8 }
	}

	#### The refuse option stays at the bottom
	option = { # refuse everything
		name = ova_champion_flavor.1.cancel
		default_option = yes
		ai_chance = { base = 1 }
		ova_champion_refuse_demand = { MOVEMENT = movement_radical }
	}
}
