ova_champion.1 = {
	type = country_event
	placement = root

	title = ova_champion.1.t
	desc = ova_champion.1.d
	flavor = ova_champion.1.f

	duration = 3

	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	trigger = {
	}

	immediate = {
		# candidates are picked HERE so the options can show their names
		ruler ?= { save_scope_as = ova_cand_ruler }
		heir ?= { save_scope_as = ova_cand_heir }
		random_interest_group = {
			limit = { exists = leader }
			leader = { save_scope_as = ova_cand_notable }
		}
	}

	option = { # scour the realm for an unknown talent
		name = ova_champion.1.opt_local
		default_option = yes
		remove_variable = ova_champion_choosing
		ova_champion_create_local = yes
	}

	option = { # the crown takes the mantle itself
		name = ova_champion.1.opt_ruler
		trigger = { exists = scope:ova_cand_ruler }
		remove_variable = ova_champion_choosing
		scope:ova_cand_ruler = { save_scope_as = ova_champion_candidate }
		ova_champion_adopt = yes
	}

	option = { # the heir must prove themselves
		name = ova_champion.1.opt_heir
		trigger = { exists = scope:ova_cand_heir }
		remove_variable = ova_champion_choosing
		scope:ova_cand_heir = { save_scope_as = ova_champion_candidate }
		ova_champion_adopt = yes
	}

	option = { # elevate a figure of consequence
		name = ova_champion.1.opt_notable
		trigger = { exists = scope:ova_cand_notable }
		remove_variable = ova_champion_choosing
		scope:ova_cand_notable = { save_scope_as = ova_champion_candidate }
		ova_champion_adopt = yes
	}
}

ova_champion.20 = { # a province carved out
	type = country_event
	placement = root
	title = ova_champion.20.t
	desc = ova_champion.20.d
	duration = 3
	event_image = { video = "unspecific_military_parade" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	trigger = { }
	immediate = {
		# prefers a state we already hold a slice of (fill it out) over a fresh
		# border state - see ova_champion_pick_seize_state
		ova_champion_pick_seize_state = yes
	}
	option = {
		name = ova_champion.20.a
		default_option = yes
		# the expedition does NOT end on a seizure - the champion presses on and
		# may carve the next province next time. The pact only ends on the
		# champion's death or when the target is annexed out of existence.
		if = {
			limit = { exists = scope:ova_seized_state }
			ova_champion_resolve_seize = yes # ONE province, not the whole state
		}
		if = {
			limit = { has_variable = ova_province_seized }
			remove_variable = ova_province_seized
			# ?= - the raided region_state may cease to exist if the carved
			# province was its last
			scope:ova_seized_state ?= { add_devastation = 20 }
			ova_champion_add_xp = { AMOUNT = 40 }
			ova_champion_add_popularity = { AMOUNT = 10 }
		}
		else = { # no carvable border province after all - plunder instead
			add_treasury = 250000
			ova_champion_add_xp = { AMOUNT = 40 }
			ova_champion_add_popularity = { AMOUNT = 5 }
		}
	}
}

# --------------------------- vs centralized neighbor, demonic path (30-34)
ova_champion.30 = { # a border province seized
	type = country_event
	placement = root
	title = ova_champion.30.t
	desc = ova_champion.30.d
	duration = 3
	event_image = { video = "unspecific_military_parade" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	trigger = { }
	immediate = {
		# prefers a state we already hold a slice of (fill it out); falls back
		# to a bordering state, then a coastal beachhead for a landless
		# Otherworlder crusade - see ova_champion_pick_seize_state_centralized
		ova_champion_pick_seize_state_centralized = yes
	}
	option = {
		name = ova_champion.30.a
		default_option = yes
		# the expedition presses on after a seizure (see ova_champion.20)
		if = {
			limit = { exists = scope:ova_seized_state }
			ova_champion_resolve_seize = yes # ONE adjacent province, not the whole state
		}
		if = {
			limit = { has_variable = ova_province_seized }
			remove_variable = ova_province_seized
			# ?= - the raided region_state may cease to exist if the carved
			# province was its last
			scope:ova_seized_state ?= { add_devastation = 25 }
			change_infamy = { value = 5 }
			ova_champion_add_xp = { AMOUNT = 40 }
			ova_champion_add_popularity = { AMOUNT = 10 }
			# a CENTRALIZED victim gets a claim on the region we just carved a
			# province from - a decentralized victim has no use for one, but a
			# proper nation keeps a legitimate path back to what it lost
			if = {
				limit = {
					exists = scope:ova_seized_region
					scope:ova_adventure_target ?= { NOT = { is_country_type = decentralized } }
				}
				scope:ova_seized_region = { add_claim = scope:ova_adventure_target }
			}
		}
		else = { # no carvable border province after all - plunder instead (see ova_champion.20)
			add_treasury = 250000
			ova_champion_add_xp = { AMOUNT = 40 }
			ova_champion_add_popularity = { AMOUNT = 5 }
		}
	}
}
