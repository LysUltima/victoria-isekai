#                   four private academy beats stay here as .92-.95.

namespace = ova_hero

# ------------------------------------------------------------ .1 summoning
ova_hero.1 = { # The Summoning Rite
	type = country_event
	placement = root

	title = ova_hero.1.t
	desc = ova_hero.1.d
	flavor = ova_hero.1.f

	duration = 3

	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	trigger = {
		NOT = { has_variable = ova_champion_char }
	}

	# No immediate needed: the summoned Hero's culture is one of the two
	# otherworld cultures, rolled inside ova_hero_perform_summoning, not the
	# summoner's own primary culture.

	option = { # the standard rite
		name = ova_hero.1.a
		default_option = yes
		add_treasury = { value = -10000 }
		c:LUX ?= { add_treasury = 10000 }
        custom_tooltip = {
            text = ova_hero_perform_summoning_tt
            ova_hero_perform_summoning = { LEVEL = 0 }
        }
        ai_chance = {
			base = 10
        }
	}

	option = { # the grand rite - a second fee, buys a stronger Hero
		name = ova_hero.1.b
		add_treasury = { value = -100000 }
		c:LUX ?= { add_treasury = 100000 }
		custom_tooltip = {
            text = ova_hero_perform_summoning_better_tt
            ova_hero_perform_summoning = { LEVEL = 15 }
        }
        ai_chance = {
			base = 10
            modifier = {
				trigger = {
					country_rank > rank_value:major_power
				}
				add = 20
			}
        }
	}

	option = { # we hesitate - the fee is already spent regardless
		name = ova_hero.1.c
        trigger = { is_player = yes }
		remove_variable = ova_hero_summoning_pending
        ai_chance = {
			base = 0
        }
	}
}

# ================== .90-.109 Victoria's arc, the drip-feed beats ==================
# The chapters of Victoria's chain are strict chains: every beat below requires
# the previous beat's flag and refuses to fire twice, so a chapter walks its
# story one step at a time instead of resolving in a single lucky roll. The
# pulse-driven chapters (1, 3, 4, 6, 8) offer their next beat through
# effect_ova_victoria_chN_tick (chapters 1/6/8 three times a year, 3/4 twice);
# the academy (chapter 2) is scheduled by day offset from the enrollment
# decision, and the ward (5), archive (7) and vault (9) chain off their own
# buttons. The 2026-07-17 beats live at .120-.126 below.
#   .90-.91   chapter 1, the hall goes up around her (with .120/.121)
#   .92-.95   chapter 2, the middle years at Saint Christine's
#   .96       chapter 3, the wall will not move on its own (with .122/.123)
#   .97-.98   chapter 4, the dusk country before the grey wings
#   .99-.100  chapter 5, the arithmetic before the ledger (then .124)
#   .101-.102 chapter 6, an enemy becomes a friend by degrees (then .125)
#   .103-.104 chapter 7, the shelves before the amended word (then .126)
#   .105-.107 chapter 8, the court learns her face
#   .108-.109 chapter 9, the vault before the whole of it
