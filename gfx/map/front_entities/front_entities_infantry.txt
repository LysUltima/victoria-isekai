military_infantry_european_irregular_diorama_entity = {
	entity = "military_infantry_european_irregular_entity"

	group = "infantry"
	
	trigger = {
		scope:culture = {
			OR = {
				has_culture_graphics = european
				has_culture_graphics = french
				has_culture_graphics = brazilian
				has_culture_graphics = balkan
			}
		}
		scope:country = { 
			OR = {
				NOT = { always = no } # vanilla ref 'c:SPC' not present in this mod 
				AND = {
					always = no # vanilla ref 'c:SPC' not present in this mod
					NOT = { has_dlc_feature = ip4_cosmetics }
				}
			}
		}
		scope:combat_unit_type = unit_type:combat_unit_type_irregular_infantry
	}
}
